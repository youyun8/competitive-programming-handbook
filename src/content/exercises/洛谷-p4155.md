solution_outline: 正規化並排序 2N 區間。因無包含關係，右端隨排序遞增；雙指標建最遠後繼，再建 jump[k]。每個原區間由大次方跳躍，計數到右端首次覆蓋一圈。
proof_or_invariant: 交換論證可將任一最優接力的下一人換成可接觸且右端最遠者而不增加人數，因此反覆 next 最優。複製後覆蓋一圈等價於右端到達 start+M；倍增只壓縮相同 next 序列，所得最少跳數正確。

  #include <bits/stdc++.h>
  using namespace std;
  struct Interval{long long left,right;int id;};
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int n;long long circumference;cin>>n>>circumference;
      vector<Interval> intervals(static_cast<size_t>(2*n));
      for(int i=0;i<n;++i){long long l,r;cin>>l>>r;if(r<l)r+=circumference;intervals[static_cast<size_t>(i)]={l,r,i};}
      sort(intervals.begin(),intervals.begin()+n,[](const auto&a,const auto&b){return a.left<b.left;});
      for(int i=0;i<n;++i){const auto&x=intervals[static_cast<size_t>(i)];intervals[static_cast<size_t>(i+n)]={x.left+circumference,x.right+circumference,x.id};}
      int levels=1;while((1<<levels)<=2*n)++levels;
      vector<vector<int>> jump(static_cast<size_t>(levels),vector<int>(static_cast<size_t>(2*n)));
      int farthest=0;
      for(int i=0;i<2*n;++i){farthest=max(farthest,i);while(farthest+1<2*n&&intervals[static_cast<size_t>(farthest+1)].left<=intervals[static_cast<size_t>(i)].right)++farthest;jump[0][static_cast<size_t>(i)]=farthest;}
      for(int k=1;k<levels;++k)for(int i=0;i<2*n;++i)jump[static_cast<size_t>(k)][static_cast<size_t>(i)]=jump[static_cast<size_t>(k-1)][static_cast<size_t>(jump[static_cast<size_t>(k-1)][static_cast<size_t>(i)])];
      vector<int> answer(static_cast<size_t>(n));
      for(int start=0;start<n;++start){const long long target=intervals[static_cast<size_t>(start)].left+circumference;if(intervals[static_cast<size_t>(start)].right>=target){answer[static_cast<size_t>(intervals[static_cast<size_t>(start)].id)]=1;continue;}int current=start,count=1;for(int k=levels-1;k>=0;--k){int next=jump[static_cast<size_t>(k)][static_cast<size_t>(current)];if(next>current&&intervals[static_cast<size_t>(next)].right<target){current=next;count+=1<<k;}}answer[static_cast<size_t>(intervals[static_cast<size_t>(start)].id)]=count+1;}
      for(int i=0;i<n;++i)cout<<answer[static_cast<size_t>(i)]<<(i+1==n?'\n':' ');
  }
