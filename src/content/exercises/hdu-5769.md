proof_or_invariant: 依 SA 掃描時，後綴 p 的前 lcp 個前綴與前一後綴相同，恰是已出現的內容；其餘前綴各代表一個新的不同子字串。前綴含 X 當且僅當長度至少為 next_x[p]-p+1，因此取兩個下界的最大值後，剩餘前綴恰好一一對應所有新合法子字串。

cpp_solution: |
  #include <algorithm>
  #include <iostream>
  #include <numeric>
  #include <string>
  #include <vector>
  using namespace std;
  static vector<int> suffix_array(const string& text) {
      const int n = static_cast<int>(text.size());
      vector<int> sa(static_cast<size_t>(n)), rank_of(static_cast<size_t>(n)), next_rank(static_cast<size_t>(n));
      iota(sa.begin(), sa.end(), 0);
      for (int i = 0; i < n; ++i) rank_of[static_cast<size_t>(i)] = text[static_cast<size_t>(i)] - 'a';
      for (int length = 1;; length <<= 1) {
          sort(sa.begin(), sa.end(), [&](int x, int y) {
              const pair<int, int> left{rank_of[static_cast<size_t>(x)],
                  x + length < n ? rank_of[static_cast<size_t>(x + length)] : -1};
              const pair<int, int> right{rank_of[static_cast<size_t>(y)],
                  y + length < n ? rank_of[static_cast<size_t>(y + length)] : -1};
              return left < right;
          });
          next_rank[static_cast<size_t>(sa[0])] = 0;
          for (int i = 1; i < n; ++i) {
              const int x = sa[static_cast<size_t>(i - 1)], y = sa[static_cast<size_t>(i)];
              const pair<int, int> left{rank_of[static_cast<size_t>(x)],
                  x + length < n ? rank_of[static_cast<size_t>(x + length)] : -1};
              const pair<int, int> right{rank_of[static_cast<size_t>(y)],
                  y + length < n ? rank_of[static_cast<size_t>(y + length)] : -1};
              next_rank[static_cast<size_t>(y)] = next_rank[static_cast<size_t>(x)] + (left != right);
          }
          rank_of = next_rank;
          if (rank_of[static_cast<size_t>(sa.back())] == n - 1) break;
      }
      return sa;
  }
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int tests = 0;
      cin >> tests;
      for (int test = 1; test <= tests; ++test) {
          char wanted = '\0';
          string text;
          cin >> wanted >> text;
          const int n = static_cast<int>(text.size());
          const vector<int> sa = suffix_array(text);
          vector<int> rank_of(static_cast<size_t>(n)), lcp(static_cast<size_t>(n));
          for (int i = 0; i < n; ++i) rank_of[static_cast<size_t>(sa[static_cast<size_t>(i)])] = i;
          for (int i = 0, matched = 0; i < n; ++i) {
              const int rank = rank_of[static_cast<size_t>(i)];
              if (rank == 0) continue;
              const int previous = sa[static_cast<size_t>(rank - 1)];
              while (i + matched < n && previous + matched < n &&
                     text[static_cast<size_t>(i + matched)] == text[static_cast<size_t>(previous + matched)]) ++matched;
              lcp[static_cast<size_t>(rank)] = matched;
              if (matched > 0) --matched;
          }
          vector<int> next_wanted(static_cast<size_t>(n), n);
          int next = n;
          for (int i = n - 1; i >= 0; --i) {
              if (text[static_cast<size_t>(i)] == wanted) next = i;
              next_wanted[static_cast<size_t>(i)] = next;
          }
          long long answer = 0;
          for (int i = 0; i < n; ++i) {
              const int start = sa[static_cast<size_t>(i)];
              if (next_wanted[static_cast<size_t>(start)] == n) continue;
              const int excluded = max(lcp[static_cast<size_t>(i)],
                                       next_wanted[static_cast<size_t>(start)] - start);
              answer += n - start - excluded;
          }
          cout << "Case #" << test << ": " << answer << '\n';
      }
      return 0;
  }
