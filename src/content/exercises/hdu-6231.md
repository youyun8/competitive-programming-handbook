cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int tests; cin >> tests;
      while (tests--) {
          int n, k; long long m; cin >> n >> k >> m;
          vector<int> a(static_cast<size_t>(n)), values;
          for (int& x : a) { cin >> x; values.push_back(x); }
          sort(values.begin(), values.end()); values.erase(unique(values.begin(), values.end()), values.end());
          const auto feasible = [&](int threshold) {
              long long intervals = 0; int right = 0, count = 0;
              for (int left = 0; left < n; ++left) {
                  while (right < n && count < k) { if (a[static_cast<size_t>(right)] >= threshold) ++count; ++right; }
                  if (count < k) break;
                  intervals += static_cast<long long>(n - right + 1);
                  if (a[static_cast<size_t>(left)] >= threshold) --count;
                  if (intervals >= m) return true;
              }
              return intervals >= m;
          };
          size_t low = 0, high = values.size();
          while (low + 1 < high) {
              const size_t mid = low + (high - low) / 2;
              if (feasible(values[mid])) low = mid; else high = mid;
          }
          cout << values[low] << '\n';
      }
  }
