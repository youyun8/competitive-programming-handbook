cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  using U64 = uint64_t;
  static U64 add_mod(U64 a, U64 b, U64 mod) { return a >= mod - b ? a - (mod - b) : a + b; }
  static U64 mul_mod(U64 a, U64 b, U64 mod) {
      U64 result = 0; a %= mod;
      while (b > 0) {
          if ((b & 1U) != 0U) result = add_mod(result, a, mod);
          a = add_mod(a, a, mod); b >>= 1U;
      }
      return result;
  }
  static U64 power_mod(U64 base, U64 exponent, U64 mod) {
      U64 result = 1;
      while (exponent > 0) {
          if ((exponent & 1U) != 0U) result = mul_mod(result, base, mod);
          base = mul_mod(base, base, mod); exponent >>= 1U;
      }
      return result;
  }
  static bool is_prime(U64 n) {
      if (n < 2) return false;
      for (U64 p : {2U, 3U, 5U, 7U, 11U, 13U, 17U, 19U, 23U, 29U, 31U, 37U})
          if (n % p == 0) return n == p;
      U64 odd = n - 1; int twos = 0;
      while ((odd & 1U) == 0U) { odd >>= 1U; ++twos; }
      for (U64 base : {2U, 3U, 5U, 7U, 11U, 13U, 17U, 19U, 23U, 29U, 31U, 37U}) {
          U64 value = power_mod(base, odd, n);
          if (value == 1 || value == n - 1) continue;
          bool passed = false;
          for (int i = 1; i < twos; ++i) {
              value = mul_mod(value, value, n);
              if (value == n - 1) { passed = true; break; }
          }
          if (!passed) return false;
      }
      return true;
  }
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int tests; cin >> tests;
      while (tests-- > 0) {
          U64 p; cin >> p;
          U64 q = p - 1;
          while (!is_prime(q)) --q;
          U64 tail = 1;
          for (U64 value = q + 1; value < p; ++value) tail = mul_mod(tail, value, p);
          cout << mul_mod(p - 1, power_mod(tail, p - 2, p), p) << '\n';
      }
      return 0;
  }
