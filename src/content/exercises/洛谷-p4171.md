cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int tests;
      if (!(cin >> tests)) { return 0; }
      while (tests-- > 0) {
          int n, m;
          cin >> n >> m;
          vector<vector<int>> graph(static_cast<size_t>(2 * n));
          auto literal = [](const string& token) {
              const int variable = stoi(token.substr(1)) - 1;
              const int value = token[0] == 'm' ? 1 : 0;
              return 2 * variable + value;
          };
          for (int i = 0; i < m; ++i) {
              string left_token, right_token;
              cin >> left_token >> right_token;
              const int left = literal(left_token);
              const int right = literal(right_token);
              graph[static_cast<size_t>(left ^ 1)].push_back(right);
              graph[static_cast<size_t>(right ^ 1)].push_back(left);
          }
          vector<int> dfn(static_cast<size_t>(2 * n), 0), low(static_cast<size_t>(2 * n), 0);
          vector<int> component(static_cast<size_t>(2 * n), -1), stack_nodes;
          vector<char> on_stack(static_cast<size_t>(2 * n), 0);
          int timer = 0, component_count = 0;
          auto tarjan = [&](auto&& self, int node) -> void {
              dfn[static_cast<size_t>(node)] = low[static_cast<size_t>(node)] = ++timer;
              stack_nodes.push_back(node);
              on_stack[static_cast<size_t>(node)] = 1;
              for (const int next : graph[static_cast<size_t>(node)]) {
                  if (dfn[static_cast<size_t>(next)] == 0) {
                      self(self, next);
                      low[static_cast<size_t>(node)] =
                          min(low[static_cast<size_t>(node)], low[static_cast<size_t>(next)]);
                  } else if (on_stack[static_cast<size_t>(next)]) {
                      low[static_cast<size_t>(node)] =
                          min(low[static_cast<size_t>(node)], dfn[static_cast<size_t>(next)]);
                  }
              }
              if (low[static_cast<size_t>(node)] != dfn[static_cast<size_t>(node)]) { return; }
              while (true) {
                  const int top = stack_nodes.back();
                  stack_nodes.pop_back();
                  on_stack[static_cast<size_t>(top)] = 0;
                  component[static_cast<size_t>(top)] = component_count;
                  if (top == node) { break; }
              }
              ++component_count;
          };
          for (int node = 0; node < 2 * n; ++node) {
              if (dfn[static_cast<size_t>(node)] == 0) { tarjan(tarjan, node); }
          }
          bool possible = true;
          for (int i = 0; i < n; ++i) {
              if (component[static_cast<size_t>(2 * i)] ==
                  component[static_cast<size_t>(2 * i + 1)]) {
                  possible = false;
              }
          }
          cout << (possible ? "GOOD" : "BAD") << '\n';
      }
      return 0;
  }
