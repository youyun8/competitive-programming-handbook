statement: 給定若干小寫詞根，求長度介於 1 與 L 的小寫字串中，至少包含一個詞根者有多少個；答案按模 2^64 的自然溢位計算。

output_format: 每組輸出一行答案，視為無號 64 位整數。

  - input: "2 3\naa ab\n1 2\na\n"
    output: "104\n52"
    explanation: 第一組長度不超過 3 且含 aa 或 ab 的字串共 104 個；第二組共 52 個。
core_knowledge: [AC 自動機安全狀態, 矩陣快速冪, 補集計數]
judgment: 長度 1 到 L 全部納入；重疊詞根仍只判斷字串是否至少命中一次；運算等同 modulo 2^64。
hints:
  - AC 節點若本身或 fail 祖先是詞根終點，就不能出現在「完全不含詞根」的路徑中。
  - 只保留安全狀態的字元轉移，矩陣冪可計算固定長度的安全字串數。
  - 在矩陣增加累加維度，同時計算長度 1..L 的總數與安全數；兩者相減即答案。
solution_outline: 建 AC 自動機並沿 fail 傳遞 forbidden 標記。以安全節點為矩陣狀態，字元轉移只在目的節點安全時累加；增加一個恆等累加狀態，取矩陣的 L+1 次方得到長度 0..L 的安全路徑和。另用二維矩陣求 26^0+...+26^L，兩者都扣除空字串後相減。
proof_or_invariant: AC 狀態完整記錄目前字串後綴中可能匹配詞根的資訊，因此路徑始終停在安全節點，當且僅當整個字串不含任何詞根。矩陣乘法枚舉所有逐字轉移，累加維度保存各長度路徑數之和；故補集正是至少命中一個詞根者。
common_errors:
  - 未把 fail 祖先的終點標記傳給目前節點
  - 只計長度恰為 L 而漏掉較短字串
  - 用有號溢位實作 modulo 2^64

cpp_solution: |
  #include <array>
  #include <cstdint>
  #include <iostream>
  #include <queue>
  #include <string>
  #include <vector>
  using namespace std;
  using U64 = uint64_t;
  using Matrix = vector<vector<U64>>;
  struct Node { array<int, 26> next{}; int fail = 0; bool forbidden = false; };
  static Matrix multiply(const Matrix& a, const Matrix& b) {
      const size_t n = a.size();
      Matrix c(n, vector<U64>(n));
      for (size_t i = 0; i < n; ++i) for (size_t k = 0; k < n; ++k) {
          if (a[i][k] == 0) continue;
          for (size_t j = 0; j < n; ++j) c[i][j] += a[i][k] * b[k][j];
      }
      return c;
  }
  static Matrix power(Matrix base, uint64_t exponent) {
      Matrix result(base.size(), vector<U64>(base.size()));
      for (size_t i = 0; i < base.size(); ++i) result[i][i] = 1;
      while (exponent > 0) {
          if ((exponent & 1U) != 0U) result = multiply(result, base);
          base = multiply(base, base);
          exponent >>= 1U;
      }
      return result;
  }
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int pattern_count = 0;
      uint64_t max_length = 0;
      while (cin >> pattern_count >> max_length) {
          vector<Node> trie(1);
          for (int i = 0; i < pattern_count; ++i) {
              string word;
              cin >> word;
              int node = 0;
              for (char ch : word) {
                  const size_t letter = static_cast<size_t>(ch - 'a');
                  if (trie[static_cast<size_t>(node)].next[letter] == 0) {
                      trie[static_cast<size_t>(node)].next[letter] = static_cast<int>(trie.size());
                      trie.push_back(Node{});
                  }
                  node = trie[static_cast<size_t>(node)].next[letter];
              }
              trie[static_cast<size_t>(node)].forbidden = true;
          }
          queue<int> pending;
          for (size_t c = 0; c < 26; ++c) if (trie[0].next[c] != 0) pending.push(trie[0].next[c]);
          while (!pending.empty()) {
              const int node = pending.front();
              pending.pop();
              Node& current = trie[static_cast<size_t>(node)];
              current.forbidden = current.forbidden || trie[static_cast<size_t>(current.fail)].forbidden;
              for (size_t c = 0; c < 26; ++c) {
                  int& child = current.next[c];
                  if (child != 0) {
                      trie[static_cast<size_t>(child)].fail = trie[static_cast<size_t>(current.fail)].next[c];
                      pending.push(child);
                  } else {
                      child = trie[static_cast<size_t>(current.fail)].next[c];
                  }
              }
          }
          const size_t states = trie.size();
          Matrix safe(states + 1, vector<U64>(states + 1));
          for (size_t node = 0; node < states; ++node) if (!trie[node].forbidden) {
              for (size_t c = 0; c < 26; ++c) {
                  const int next = trie[node].next[c];
                  if (!trie[static_cast<size_t>(next)].forbidden) ++safe[node][static_cast<size_t>(next)];
              }
              safe[node][states] = 1;
          }
          safe[states][states] = 1;
          const U64 safe_sum = power(safe, max_length + 1)[0][states] - 1;
          Matrix all{{26, 1}, {0, 1}};
          const U64 all_sum = power(all, max_length + 1)[0][1] - 1;
          cout << all_sum - safe_sum << '\n';
      }
      return 0;
  }
