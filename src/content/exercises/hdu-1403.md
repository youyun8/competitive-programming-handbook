proof_or_invariant: 任一同時出現在兩字串的子字串，都是一個 a 後綴與一個 b 後綴的公共前綴。對使其 LCP 最大的一對異源後綴，在 SA 中兩來源區塊的交界必有一對相鄰異源後綴，其 LCP 不小於該值；反之每個被計入的 LCP 顯然是兩字串的公共子字串，故最大值相等。

cpp_solution: |
  #include <algorithm>
  #include <iostream>
  #include <numeric>
  #include <string>
  #include <vector>
  using namespace std;
  static vector<int> suffix_array(const string& text) {
      const int n = static_cast<int>(text.size());
      vector<int> sa(static_cast<size_t>(n)), rank_of(static_cast<size_t>(n)), next_rank(static_cast<size_t>(n));
      iota(sa.begin(), sa.end(), 0);
      for (int i = 0; i < n; ++i) rank_of[static_cast<size_t>(i)] = static_cast<unsigned char>(text[static_cast<size_t>(i)]);
      for (int length = 1;; length <<= 1) {
          sort(sa.begin(), sa.end(), [&](int x, int y) {
              if (rank_of[static_cast<size_t>(x)] != rank_of[static_cast<size_t>(y)])
                  return rank_of[static_cast<size_t>(x)] < rank_of[static_cast<size_t>(y)];
              const int xr = x + length < n ? rank_of[static_cast<size_t>(x + length)] : -1;
              const int yr = y + length < n ? rank_of[static_cast<size_t>(y + length)] : -1;
              return xr < yr;
          });
          next_rank[static_cast<size_t>(sa[0])] = 0;
          for (int i = 1; i < n; ++i) {
              const int x = sa[static_cast<size_t>(i - 1)], y = sa[static_cast<size_t>(i)];
              const pair<int, int> left{rank_of[static_cast<size_t>(x)],
                  x + length < n ? rank_of[static_cast<size_t>(x + length)] : -1};
              const pair<int, int> right{rank_of[static_cast<size_t>(y)],
                  y + length < n ? rank_of[static_cast<size_t>(y + length)] : -1};
              next_rank[static_cast<size_t>(y)] = next_rank[static_cast<size_t>(x)] + (left != right);
          }
          rank_of = next_rank;
          if (rank_of[static_cast<size_t>(sa.back())] == n - 1) break;
      }
      return sa;
  }
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      string first, second;
      while (cin >> first >> second) {
          const int split = static_cast<int>(first.size());
          const string text = first + char(1) + second;
          const int n = static_cast<int>(text.size());
          const vector<int> sa = suffix_array(text);
          vector<int> rank_of(static_cast<size_t>(n)), lcp(static_cast<size_t>(n));
          for (int i = 0; i < n; ++i) rank_of[static_cast<size_t>(sa[static_cast<size_t>(i)])] = i;
          for (int i = 0, matched = 0; i < n; ++i) {
              const int rank = rank_of[static_cast<size_t>(i)];
              if (rank == 0) continue;
              const int previous = sa[static_cast<size_t>(rank - 1)];
              while (i + matched < n && previous + matched < n &&
                     text[static_cast<size_t>(i + matched)] == text[static_cast<size_t>(previous + matched)]) ++matched;
              lcp[static_cast<size_t>(rank)] = matched;
              if (matched > 0) --matched;
          }
          int answer = 0;
          auto side = [split](int position) { return position < split ? 0 : (position > split ? 1 : -1); };
          for (int i = 1; i < n; ++i) {
              const int left_side = side(sa[static_cast<size_t>(i - 1)]);
              const int right_side = side(sa[static_cast<size_t>(i)]);
              if (left_side >= 0 && right_side >= 0 && left_side != right_side)
                  answer = max(answer, lcp[static_cast<size_t>(i)]);
          }
          cout << answer << '\n';
      }
      return 0;
  }
