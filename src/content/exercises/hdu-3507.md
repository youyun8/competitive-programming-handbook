hints:
  - 設 s[i] 為前綴和，dp[i]=min_j(dp[j]+(s[i]-s[j])²+m)。
  - 移項後候選 j 是斜率 -2s[j]、截距 dp[j]+s[j]² 的直線，在 x=s[i] 查最小值。
  - 前綴和非遞減；用交叉乘法維護下凸殼，並特別處理相同 s 值。
solution_outline: >-
  依序計算前綴和與 dp。佇列存候選索引；查詢時比較隊首兩條線在當前 x 的值，彈掉較差者。加入新線前以交叉乘判斷尾端是否失去成為最優的可能；相同前綴和只保留截距較小者。
proof_or_invariant: >-
  展開原轉移即為所有候選直線在 x=s[i] 的最小值。斜率按加入順序單調，查詢 x 也單調，因此兩線交點一旦越過當前查詢，較差線永不再優，可從隊首刪除；尾端中間線的交點順序不遞增時也永不最優。凸殼保留且僅刪除不可能最優者。

cpp_solution: |
  #include <boost/multiprecision/cpp_int.hpp>
  #include <deque>
  #include <iostream>
  #include <vector>
  using namespace std;
  using boost::multiprecision::int128_t;
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int n;
      long long fixed_cost;
      while (cin >> n >> fixed_cost) {
          vector<long long> prefix(static_cast<size_t>(n + 1));
          vector<long long> dp(static_cast<size_t>(n + 1));
          for (int i = 1; i <= n; ++i) {
              cin >> prefix[i];
              prefix[i] += prefix[i - 1];
          }
          auto intercept = [&](int index) {
              return dp[index] + prefix[index] * prefix[index];
          };
          auto value = [&](int index, long long x) {
              return intercept(index) - 2 * prefix[index] * x;
          };
          auto obsolete = [&](int a, int b, int c) {
              return int128_t(intercept(b) - intercept(a)) *
                         (prefix[b] - prefix[c]) >=
                     int128_t(intercept(c) - intercept(b)) *
                         (prefix[a] - prefix[b]);
          };
          deque<int> hull;
          hull.push_back(0);
          for (int i = 1; i <= n; ++i) {
              while (hull.size() >= 2 &&
                     value(hull[0], prefix[i]) >= value(hull[1], prefix[i]))
                  hull.pop_front();
              dp[i] = prefix[i] * prefix[i] + fixed_cost + value(hull.front(), prefix[i]);
              if (!hull.empty() && prefix[hull.back()] == prefix[i]) {
                  if (intercept(hull.back()) <= intercept(i)) continue;
                  hull.pop_back();
              }
              while (hull.size() >= 2 &&
                     obsolete(hull[hull.size() - 2], hull.back(), i))
                  hull.pop_back();
              hull.push_back(i);
          }
          cout << dp[n] << '\n';
      }
  }
