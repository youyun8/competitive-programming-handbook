  #include <array>
  #include <iostream>
  using namespace std;
  constexpr int mod_value = 9973;
  struct Matrix {
      int size = 0;
      array<array<int, 10>, 10> value{};
  };
  Matrix multiply(const Matrix &left, const Matrix &right) {
      Matrix result;
      result.size = left.size;
      for (int i = 0; i < result.size; ++i)
          for (int k = 0; k < result.size; ++k)
              for (int j = 0; j < result.size; ++j)
                  result.value[static_cast<size_t>(i)][static_cast<size_t>(j)] =
                      (result.value[static_cast<size_t>(i)][static_cast<size_t>(j)] +
                       left.value[static_cast<size_t>(i)][static_cast<size_t>(k)] *
                           right.value[static_cast<size_t>(k)][static_cast<size_t>(j)]) %
                      mod_value;
      return result;
  }
  Matrix matrix_power(Matrix base, int exponent) {
      Matrix result;
      result.size = base.size;
      for (int i = 0; i < result.size; ++i)
          result.value[static_cast<size_t>(i)][static_cast<size_t>(i)] = 1;
      while (exponent > 0) {
          if ((exponent & 1) != 0) result = multiply(result, base);
          base = multiply(base, base);
          exponent >>= 1;
      }
      return result;
  }
  int scalar_power(int base, int exponent) {
      int result = 1;
      while (exponent > 0) {
          if ((exponent & 1) != 0) result = result * base % mod_value;
          base = base * base % mod_value;
          exponent >>= 1;
      }
      return result;
  }
  int totient(int value) {
      int result = value;
      for (int p = 2; static_cast<long long>(p) * p <= value; ++p) {
          if (value % p != 0) continue;
          result -= result / p;
          while (value % p == 0) value /= p;
      }
      if (value > 1) result -= result / value;
      return result;
  }
  int trace_power(const Matrix &matrix, int exponent) {
      const Matrix powered = matrix_power(matrix, exponent);
      int trace = 0;
      for (int i = 0; i < powered.size; ++i)
          trace = (trace + powered.value[static_cast<size_t>(i)][static_cast<size_t>(i)]) % mod_value;
      return trace;
  }
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int test_cases;
      cin >> test_cases;
      while (test_cases-- > 0) {
          int n, color_count, forbidden_count;
          cin >> n >> color_count >> forbidden_count;
          Matrix adjacency;
          adjacency.size = color_count;
          for (int i = 0; i < color_count; ++i)
              for (int j = 0; j < color_count; ++j)
                  adjacency.value[static_cast<size_t>(i)][static_cast<size_t>(j)] = 1;
          while (forbidden_count-- > 0) {
              int first, second;
              cin >> first >> second;
              --first; --second;
              adjacency.value[static_cast<size_t>(first)][static_cast<size_t>(second)] = 0;
              adjacency.value[static_cast<size_t>(second)][static_cast<size_t>(first)] = 0;
          }
          long long fixed_sum = 0;
          for (int divisor = 1; static_cast<long long>(divisor) * divisor <= n; ++divisor) {
              if (n % divisor != 0) continue;
              const int other = n / divisor;
              fixed_sum += static_cast<long long>(totient(other) % mod_value) *
                           trace_power(adjacency, divisor);
              if (divisor != other)
                  fixed_sum += static_cast<long long>(totient(divisor) % mod_value) *
                               trace_power(adjacency, other);
              fixed_sum %= mod_value;
          }
          cout << fixed_sum * scalar_power(n % mod_value, mod_value - 2) % mod_value << '\n';
      }
      return 0;
  }
