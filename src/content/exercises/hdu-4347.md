  #if defined(__GNUC__)
  #pragma GCC diagnostic ignored "-Wconversion"
  #pragma GCC diagnostic ignored "-Wshadow"
  #pragma GCC diagnostic ignored "-Wpedantic"
  #pragma GCC diagnostic ignored "-Wsign-compare"
  #pragma GCC diagnostic ignored "-Wunused-parameter"

  #pragma GCC diagnostic ignored "-Wimplicit-fallthrough"

  #include <algorithm>
  #include <iostream>
  #include <limits>
  #include <queue>
  #include <utility>
  #include <vector>
  using namespace std;

  struct Point {
      vector<long long> coordinate;
  };
  struct Node {
      Point point;
      int child[2]{};
      vector<long long> minimum, maximum;
  };
  int dimensions;
  vector<Point> points;
  vector<Node> tree;

  void pull(int node) {
      tree[node].minimum = tree[node].maximum = tree[node].point.coordinate;
      for (int side = 0; side < 2; ++side) {
          int next = tree[node].child[side];
          if (next == 0) continue;
          for (int dimension = 0; dimension < dimensions; ++dimension) {
              tree[node].minimum[static_cast<size_t>(dimension)] =
                  min(tree[node].minimum[static_cast<size_t>(dimension)],
                      tree[next].minimum[static_cast<size_t>(dimension)]);
              tree[node].maximum[static_cast<size_t>(dimension)] =
                  max(tree[node].maximum[static_cast<size_t>(dimension)],
                      tree[next].maximum[static_cast<size_t>(dimension)]);
          }
      }
  }
  int build(int left, int right, int dimension) {
      if (left >= right) return 0;
      int middle = (left + right) / 2;
      nth_element(points.begin() + left, points.begin() + middle, points.begin() + right,
                  [dimension](const Point& first, const Point& second) {
                      return first.coordinate[static_cast<size_t>(dimension)] <
                             second.coordinate[static_cast<size_t>(dimension)];
                  });
      int node = middle + 1;
      tree[node].point = points[static_cast<size_t>(middle)];
      tree[node].child[0] = build(left, middle, (dimension + 1) % dimensions);
      tree[node].child[1] = build(middle + 1, right, (dimension + 1) % dimensions);
      pull(node);
      return node;
  }
  long long square(long long value) { return value * value; }
  long long point_distance(const Point& first, const Point& second) {
      long long answer = 0;
      for (int dimension = 0; dimension < dimensions; ++dimension)
          answer += square(first.coordinate[static_cast<size_t>(dimension)] -
                           second.coordinate[static_cast<size_t>(dimension)]);
      return answer;
  }
  long long box_distance(int node, const Point& query) {
      long long answer = 0;
      for (int dimension = 0; dimension < dimensions; ++dimension) {
          long long value = query.coordinate[static_cast<size_t>(dimension)];
          if (value < tree[node].minimum[static_cast<size_t>(dimension)])
              answer += square(tree[node].minimum[static_cast<size_t>(dimension)] - value);
          else if (value > tree[node].maximum[static_cast<size_t>(dimension)])
              answer += square(value - tree[node].maximum[static_cast<size_t>(dimension)]);
      }
      return answer;
  }
  using Candidate = pair<long long, int>;
  void search(int node, const Point& query, int wanted, priority_queue<Candidate>& candidates) {
      if (node == 0) return;
      long long candidate_distance = point_distance(tree[node].point, query);
      if (static_cast<int>(candidates.size()) < wanted) {
          candidates.push({candidate_distance, node});
      } else if (candidate_distance < candidates.top().first) {
          candidates.pop();
          candidates.push({candidate_distance, node});
      }
      int first = tree[node].child[0], second = tree[node].child[1];
      long long first_bound = first == 0 ? numeric_limits<long long>::max() : box_distance(first, query);
      long long second_bound = second == 0 ? numeric_limits<long long>::max() : box_distance(second, query);
      if (first_bound > second_bound) {
          swap(first, second);
          swap(first_bound, second_bound);
      }
      auto useful = [&](long long bound) {
          return static_cast<int>(candidates.size()) < wanted || bound <= candidates.top().first;
      };
      if (first != 0 && useful(first_bound)) search(first, query, wanted, candidates);
      if (second != 0 && useful(second_bound)) search(second, query, wanted, candidates);
  }
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int n = 0;
      while (cin >> n >> dimensions) {
          points.assign(static_cast<size_t>(n), {});
          for (Point& point : points) {
              point.coordinate.resize(static_cast<size_t>(dimensions));
              for (long long& value : point.coordinate) cin >> value;
          }
          tree.assign(static_cast<size_t>(n) + 1U, {});
          int root = build(0, n, 0);
          int query_count = 0;
          cin >> query_count;
          while (query_count-- > 0) {
              Point query;
              query.coordinate.resize(static_cast<size_t>(dimensions));
              for (long long& value : query.coordinate) cin >> value;
              int wanted = 0;
              cin >> wanted;
              priority_queue<Candidate> candidates;
              search(root, query, wanted, candidates);
              vector<Point> answer;
              while (!candidates.empty()) {
                  answer.push_back(tree[candidates.top().second].point);
                  candidates.pop();
              }
              reverse(answer.begin(), answer.end());
              cout << "the closest " << wanted << " points are:\n";
              for (const Point& point : answer) {
                  for (int dimension = 0; dimension < dimensions; ++dimension) {
                      if (dimension != 0) cout << ' ';
                      cout << point.coordinate[static_cast<size_t>(dimension)];
                  }
                  cout << '\n';
              }
          }
      }
      return 0;
  }
