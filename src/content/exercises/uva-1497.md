statement: 給定一段操作三維點的程式與 n 個初始點，求程式執行後各點座標。程式可平移、沿三軸分別縮放、繞通過原點的任意軸旋轉，並可用 repeat k ... end 巢狀重複一段操作。
constraints:
  - 每組 '1 <= n <= 1000'，測資不超過 20 組
  - 程式少於 100 行，且 repeat 可巢狀
  - 'k >= 0 且可存於 32-bit signed integer'
  - 除 n、k 外的數值皆為絕對值不超過 1000 的浮點數
  - 單獨一個 n=0 結束
input_format: 每組先給點數 n，接著是一段以未配對 end 結束的合法程式，再給 n 行三維點。指令為 translate tx ty tz、scale a b c、rotate a b c d、repeat k 與 end；相鄰測資可能以空白行分隔。
output_format: 依輸入順序輸出每個點變換後的 x、y、z，四捨五入至小數點後兩位；每組輸出後再印一個空白行。
samples:
  - input: |
      2
      rotate 1.0 1.0 1.0 90.0
      translate 2.0 2.0 2.0
      end
      1.0 1.0 1.0
      1.0 0.0 0.0
      0
    output: |
      3.00 3.00 3.00
      2.33 2.91 1.76

    explanation: 軸上的 (1,1,1) 旋轉後不變，再平移成 (3,3,3)；第二點先依指定方向繞 (1,1,1) 軸轉 90 度，再三軸各加 2。
core_knowledge:
  - 4×4 齊次矩陣可統一表示三維平移、縮放與旋轉
  - 連續操作以矩陣乘法合成，次序不可顛倒
  - repeat 區塊等於區塊矩陣的 k 次方
judgment: 點數可達 1000、k 可很大，逐點逐次執行會超時。先遞迴解析程式並合成一個仿射矩陣，repeat 用二進位快速冪，最後每點只乘一次矩陣。
hints:
  - 使用齊次向量 (x,y,z,1)，平移就能和其他操作一起寫成 4×4 矩陣。
  - 若目前合成矩陣為 M，下一個指令矩陣為 A，按輸入順序執行後應更新成 A·M。
  - 遇到 repeat k 時遞迴解析到配對 end，將整段矩陣快速冪成 B^k，再像一個普通指令般合成。
solution_outline: 以遞迴函式讀到當層 end，逐項左乘合成矩陣。平移與縮放直接建矩陣；旋轉將軸正規化後使用 Rodrigues 公式；repeat 對遞迴所得矩陣做快速冪。最外層矩陣完成後變換所有點。
proof_or_invariant: 解析函式處理完任意前綴後，累積矩陣 M 滿足「對任意輸入點，M 乘其齊次向量恰等於依序執行此前綴的結果」。普通指令 A 使結果成為 A(Mp)，故左乘維持不變量。遞迴依相同不變量得到區塊矩陣 B，重複 k 次即 B^k，快速冪只利用矩陣乘法結合律而保持相等。結束時 M 因此等價於完整程式。
complexity:
  time: O(L log K + n)，L 為程式行數，log K 表示各 repeat 快速冪成本總和；4×4 矩陣運算視為常數
  space: O(D)，D 為 repeat 最大巢狀深度
common_errors:
  - 把新指令矩陣乘在累積矩陣右側，導致操作順序反轉
  - 使用 3×3 矩陣而無法同時合成平移
  - 把題目旋轉方向解讀成相反號
  - repeat k 逐次相乘，k 很大時超時
  - 輸出接近零的數成為 -0.00 或漏掉測資間空白行

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;

  struct Matrix {
      array<array<long double, 4>, 4> value{};
  };

  static Matrix identity() {
      Matrix result{};
      for (size_t i = 0; i < 4U; ++i) { result.value[i][i] = 1.0L; }
      return result;
  }

  static Matrix operator*(const Matrix& first, const Matrix& second) {
      Matrix result{};
      for (size_t i = 0; i < 4U; ++i) {
          for (size_t k = 0; k < 4U; ++k) {
              for (size_t j = 0; j < 4U; ++j) {
                  result.value[i][j] +=
                      first.value[i][k] * second.value[k][j];
              }
          }
      }
      return result;
  }

  static Matrix matrix_power(Matrix base, int exponent) {
      Matrix result = identity();
      while (exponent > 0) {
          if ((exponent & 1) != 0) { result = result * base; }
          base = base * base;
          exponent >>= 1;
      }
      return result;
  }

  static Matrix translation(long double x, long double y, long double z) {
      Matrix result = identity();
      result.value[0][3] = x;
      result.value[1][3] = y;
      result.value[2][3] = z;
      return result;
  }

  static Matrix scaling(long double x, long double y, long double z) {
      Matrix result = identity();
      result.value[0][0] = x;
      result.value[1][1] = y;
      result.value[2][2] = z;
      return result;
  }

  static Matrix rotation(long double x, long double y, long double z,
                         long double degrees) {
      const long double length = sqrtl(x * x + y * y + z * z);
      x /= length;
      y /= length;
      z /= length;
      const long double angle = degrees * acosl(-1.0L) / 180.0L;
      const long double cosine = cosl(angle);
      const long double sine = sinl(angle);
      const long double remain = 1.0L - cosine;
      Matrix result = identity();
      result.value[0][0] = cosine + x * x * remain;
      result.value[0][1] = x * y * remain - z * sine;
      result.value[0][2] = x * z * remain + y * sine;
      result.value[1][0] = y * x * remain + z * sine;
      result.value[1][1] = cosine + y * y * remain;
      result.value[1][2] = y * z * remain - x * sine;
      result.value[2][0] = z * x * remain - y * sine;
      result.value[2][1] = z * y * remain + x * sine;
      result.value[2][2] = cosine + z * z * remain;
      return result;
  }

  static Matrix read_program() {
      Matrix accumulated = identity();
      string instruction;
      while (cin >> instruction && instruction != "end") {
          Matrix operation{};
          if (instruction == "translate") {
              long double x;
              long double y;
              long double z;
              cin >> x >> y >> z;
              operation = translation(x, y, z);
          } else if (instruction == "scale") {
              long double x;
              long double y;
              long double z;
              cin >> x >> y >> z;
              operation = scaling(x, y, z);
          } else if (instruction == "rotate") {
              long double x;
              long double y;
              long double z;
              long double degrees;
              cin >> x >> y >> z >> degrees;
              operation = rotation(x, y, z, degrees);
          } else {
              int repetitions;
              cin >> repetitions;
              operation = matrix_power(read_program(), repetitions);
          }
          accumulated = operation * accumulated;
      }
      return accumulated;
  }

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      cout << fixed << setprecision(2);
      int n;
      while (cin >> n && n != 0) {
          const Matrix transform = read_program();
          for (int i = 0; i < n; ++i) {
              array<long double, 4> point{};
              cin >> point[0] >> point[1] >> point[2];
              point[3] = 1.0L;
              array<long double, 3> result{};
              for (size_t row = 0; row < 3U; ++row) {
                  for (size_t column = 0; column < 4U; ++column) {
                      result[row] +=
                          transform.value[row][column] * point[column];
                  }
                  if (fabsl(result[row]) < 0.0005L) { result[row] = 0.0L; }
              }
              cout << result[0] << ' ' << result[1] << ' ' << result[2]
                   << '\n';
          }
          cout << '\n';
      }
      return 0;
  }
