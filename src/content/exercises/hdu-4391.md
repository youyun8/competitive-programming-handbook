statement: 一面牆由零起算的位置組成，各有整數顏色。支援把區間全部染成指定顏色，以及查詢區間內指定顏色的格數。

samples:
  - input: |
      5 4
      1 2 1 3 2
      2 0 4 1
      1 1 3 2
      2 0 4 2
      2 1 3 1
    output: |
      2
      4
      0
    explanation: 初始顏色 1 有兩格；把位置 1 到 3 全染成 2 後，除位置 0 外皆為顏色 2。

hints:
  - 每塊保存 lazy_color；不是 -1 時代表全塊同色。
  - 散塊操作前下推 lazy_color，修改後重建 unordered_map 頻率。
  - 查完整塊時若有標記直接比較，否則以 find 查頻率，避免 `map[key]` 插入無用顏色。
solution_outline: 序列分塊，完整塊使用覆蓋標記，無標記時維護塊內顏色頻率；兩端下推後暴力。
proof_or_invariant: 每塊或由 lazy 唯一描述真實顏色，或由逐點陣列與 frequency 完整描述。所有操作先維持該表示再計算，因此每格恰被修改或統計一次。
complexity:
  time: 每次操作期望 O(sqrt(n))
  space: O(n)

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main(){ios::sync_with_stdio(false);cin.tie(nullptr);int n,m;while(cin>>n>>m){int length=max(1,static_cast<int>(sqrt(static_cast<double>(n))));int blocks=(n+length-1)/length;vector<int>color(static_cast<size_t>(n)),lazy(static_cast<size_t>(blocks),-1);vector<unordered_map<int,int>>frequency(static_cast<size_t>(blocks));for(int i=0;i<n;++i){cin>>color[static_cast<size_t>(i)];++frequency[static_cast<size_t>(i/length)][color[static_cast<size_t>(i)]];}auto push=[&](int b){if(lazy[static_cast<size_t>(b)]<0)return;int right=min(n,(b+1)*length);for(int i=b*length;i<right;++i)color[static_cast<size_t>(i)]=lazy[static_cast<size_t>(b)];lazy[static_cast<size_t>(b)]=-1;};auto rebuild=[&](int b){auto&f=frequency[static_cast<size_t>(b)];f.clear();int right=min(n,(b+1)*length);for(int i=b*length;i<right;++i)++f[color[static_cast<size_t>(i)]];};while(m--){int op,l,r,z;cin>>op>>l>>r>>z;int lb=l/length,rb=r/length;if(op==1){if(lb==rb){push(lb);for(int i=l;i<=r;++i)color[static_cast<size_t>(i)]=z;rebuild(lb);}else{push(lb);for(int i=l;i<(lb+1)*length;++i)color[static_cast<size_t>(i)]=z;rebuild(lb);for(int b=lb+1;b<rb;++b){lazy[static_cast<size_t>(b)]=z;frequency[static_cast<size_t>(b)].clear();frequency[static_cast<size_t>(b)][z]=min(n,(b+1)*length)-b*length;}push(rb);for(int i=rb*length;i<=r;++i)color[static_cast<size_t>(i)]=z;rebuild(rb);}}else{int answer=0;if(lb==rb){push(lb);for(int i=l;i<=r;++i)answer+=color[static_cast<size_t>(i)]==z;}else{push(lb);for(int i=l;i<(lb+1)*length;++i)answer+=color[static_cast<size_t>(i)]==z;for(int b=lb+1;b<rb;++b){if(lazy[static_cast<size_t>(b)]>=0)answer+=(lazy[static_cast<size_t>(b)]==z)*(min(n,(b+1)*length)-b*length);else{const auto&f=frequency[static_cast<size_t>(b)];auto it=f.find(z);if(it!=f.end())answer+=it->second;}}push(rb);for(int i=rb*length;i<=r;++i)answer+=color[static_cast<size_t>(i)]==z;}cout<<answer<<'\n';}}}}
