  平行時空形成一棵發展樹。根時空只有地球；每個新時空由既有時空複製，
  再殖民一顆新星球或放棄一顆已殖民星球。星球座標為 (x,y,z)，調查費為 c。
  每次旅行指定時空 s 與固定的到達 x 座標 x0；到達點的 y、z 可自由調整。
  求選擇該時空任一已殖民星球後，航行距離平方加調查費的最小值。

  - 每個新時空的來源編號 fr 小於目前時空編號

judgment: |-
  求出每顆星球在時空 DFS 序中的所有存活區間，將其代價函數放入區間分解節點；
  每個詢問沿根到葉的凸包取最小值。
hints:
  - 因 y、z 可自由選，選星球 i 的成本可化為 `(x0-x_i)^2+c_i`。
  - 一顆星球從殖民節點的整棵子樹存活，只需扣掉所有放棄該星球的子樹。
  - 展開平方後，每顆星球是一條直線；把星球與詢問都排序，就能讓建凸包與查詢指標都單調。
solution_outline: |-
  對時空樹做 DFS 序。星球 i 的基礎存活範圍是殖民時空子樹，扣除各放棄節點子樹後
  得到若干互斥區間，全部區間總數為 O(n)。將區間拆進時空序線段樹。
  代價展開為 `x0² + (-2x_i)x0 + x_i²+c_i`。按 x_i 排序插入所有區間，
  每個線段樹節點收到斜率單調的直線，可線性建下凸包。再按 x0 排序詢問，
  每個節點用只前進的指標取最優線，沿詢問時空對應葉子的祖先取最小值。
proof_or_invariant: |-
  發展樹中一次放棄使該星球在該節點整棵子樹失效；合法性保證同一存活分支上的
  放棄子樹不會相交或巢狀，因此區間差精確描述所有存活時空。線段樹區間分解保證
  星球在且僅在其存活位置的根葉路徑上出現一次。直線斜率按 x_i 單調，
  下凸包刪除的線在任何 x0 都不可能嚴格最優；詢問 x0 單調時最優位置只會向前。
common_errors:
  - 把星球 y、z 座標也帶進距離，忽略到達點可自由調整 y、z
  - 將放棄視為全域刪除，而不是只影響該發展分支
  - 凸包叉積用 64 位元而溢位
complexity:
  time: O((n+m) log n)
  space: O(n log n)

cpp_solution: |
  #include <bits/stdc++.h>
  #include <boost/multiprecision/cpp_int.hpp>
  using namespace std;

  struct Planet {
      long long x = 0;
      long long cost = 0;
      int added_at = -1;
      bool exists = false;
  };

  struct Query {
      int universe;
      long long x;
      int id;
  };

  long long evaluate(const Planet& planet, long long x) {
      const long long difference = x - planet.x;
      return difference * difference + planet.cost;
  }

  bool redundant(int first, int second, int third,
                 const vector<Planet>& planets) {
      const long long slope_first = -2 * planets[static_cast<size_t>(first)].x;
      const long long slope_second = -2 * planets[static_cast<size_t>(second)].x;
      const long long slope_third = -2 * planets[static_cast<size_t>(third)].x;
      const long long intercept_first =
          planets[static_cast<size_t>(first)].x *
              planets[static_cast<size_t>(first)].x +
          planets[static_cast<size_t>(first)].cost;
      const long long intercept_second =
          planets[static_cast<size_t>(second)].x *
              planets[static_cast<size_t>(second)].x +
          planets[static_cast<size_t>(second)].cost;
      const long long intercept_third =
          planets[static_cast<size_t>(third)].x *
              planets[static_cast<size_t>(third)].x +
          planets[static_cast<size_t>(third)].cost;
      using boost::multiprecision::int128_t;
      return static_cast<int128_t>(intercept_second - intercept_first) *
                 (slope_second - slope_third) >=
             static_cast<int128_t>(intercept_third - intercept_second) *
                 (slope_first - slope_second);
  }

  void add_interval(int node, int l, int r, int ql, int qr, int planet,
                    vector<vector<int>>& buckets) {
      if (ql > qr || qr < l || r < ql) { return; }
      if (ql <= l && r <= qr) {
          buckets[static_cast<size_t>(node)].push_back(planet);
          return;
      }
      const int mid = (l + r) / 2;
      add_interval(node * 2, l, mid, ql, qr, planet, buckets);
      add_interval(node * 2 + 1, mid + 1, r, ql, qr, planet, buckets);
  }

  void build_hulls(vector<vector<int>>& buckets,
                   const vector<Planet>& planets) {
      for (vector<int>& bucket : buckets) {
          if (bucket.empty()) { continue; }
          vector<int> hull;
          hull.reserve(bucket.size());
          for (const int planet : bucket) {
              if (!hull.empty() &&
                  planets[static_cast<size_t>(hull.back())].x ==
                      planets[static_cast<size_t>(planet)].x) {
                  continue;
              }
              while (hull.size() >= 2 &&
                     redundant(hull[hull.size() - 2], hull.back(),
                               planet, planets)) {
                  hull.pop_back();
              }
              hull.push_back(planet);
          }
          bucket.swap(hull);
      }
  }

  long long query_path(int position, long long x, int n,
                       const vector<vector<int>>& hulls,
                       vector<size_t>& pointer,
                       const vector<Planet>& planets) {
      constexpr long long infinity = numeric_limits<long long>::max();
      long long answer = infinity;
      int node = 1;
      int l = 1;
      int r = n;
      while (true) {
          const vector<int>& hull = hulls[static_cast<size_t>(node)];
          size_t& index = pointer[static_cast<size_t>(node)];
          while (index + 1 < hull.size() &&
                 evaluate(planets[static_cast<size_t>(hull[index + 1])], x) <=
                     evaluate(planets[static_cast<size_t>(hull[index])], x)) {
              ++index;
          }
          if (!hull.empty()) {
              answer = min(
                  answer,
                  evaluate(planets[static_cast<size_t>(hull[index])], x));
          }
          if (l == r) { break; }
          const int mid = (l + r) / 2;
          if (position <= mid) {
              node *= 2;
              r = mid;
          } else {
              node = node * 2 + 1;
              l = mid + 1;
          }
      }
      return answer;
  }

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);

      int universe_count, query_count;
      long long earth_cost;
      cin >> universe_count >> query_count >> earth_cost;
      vector<vector<int>> children(static_cast<size_t>(universe_count));
      vector<vector<int>> removed_at(static_cast<size_t>(universe_count));
      vector<int> parent(static_cast<size_t>(universe_count), -1);
      vector<Planet> planets(static_cast<size_t>(universe_count));
      planets[0] = {0, earth_cost, 0, true};

      for (int universe = 1; universe < universe_count; ++universe) {
          int type, from, id;
          cin >> type >> from >> id;
          parent[static_cast<size_t>(universe)] = from;
          children[static_cast<size_t>(from)].push_back(universe);
          if (type == 0) {
              long long x, y, z, cost;
              cin >> x >> y >> z >> cost;
              static_cast<void>(y);
              static_cast<void>(z);
              planets[static_cast<size_t>(id)] =
                  {x, cost, universe, true};
          } else {
              removed_at[static_cast<size_t>(id)].push_back(universe);
          }
      }

      vector<int> tin(static_cast<size_t>(universe_count));
      vector<int> subtree_size(static_cast<size_t>(universe_count), 1);
      vector<int> order;
      order.reserve(static_cast<size_t>(universe_count));
      vector<int> stack{0};
      int timer = 0;
      while (!stack.empty()) {
          const int universe = stack.back();
          stack.pop_back();
          tin[static_cast<size_t>(universe)] = ++timer;
          order.push_back(universe);
          const auto& descendants = children[static_cast<size_t>(universe)];
          for (auto it = descendants.rbegin(); it != descendants.rend(); ++it) {
              stack.push_back(*it);
          }
      }
      for (size_t index = order.size(); index-- > 1;) {
          const int universe = order[index];
          subtree_size[static_cast<size_t>(
              parent[static_cast<size_t>(universe)])] +=
              subtree_size[static_cast<size_t>(universe)];
      }
      vector<int> tout(static_cast<size_t>(universe_count));
      for (int universe = 0; universe < universe_count; ++universe) {
          tout[static_cast<size_t>(universe)] =
              tin[static_cast<size_t>(universe)] +
              subtree_size[static_cast<size_t>(universe)] - 1;
      }

      vector<int> planet_order;
      for (int id = 0; id < universe_count; ++id) {
          if (planets[static_cast<size_t>(id)].exists) {
              planet_order.push_back(id);
          }
      }
      sort(planet_order.begin(), planet_order.end(),
           [&](int first, int second) {
               const Planet& left = planets[static_cast<size_t>(first)];
               const Planet& right = planets[static_cast<size_t>(second)];
               if (left.x != right.x) { return left.x < right.x; }
               return left.cost < right.cost;
           });

      vector<vector<int>> hulls(
          static_cast<size_t>(universe_count) * 4 + 4);
      for (const int id : planet_order) {
          const int added = planets[static_cast<size_t>(id)].added_at;
          const int left = tin[static_cast<size_t>(added)];
          const int right = tout[static_cast<size_t>(added)];
          vector<int>& removals = removed_at[static_cast<size_t>(id)];
          sort(removals.begin(), removals.end(), [&](int first, int second) {
              return tin[static_cast<size_t>(first)] <
                     tin[static_cast<size_t>(second)];
          });
          int cursor = left;
          for (const int removed : removals) {
              const int removed_left = tin[static_cast<size_t>(removed)];
              const int removed_right = tout[static_cast<size_t>(removed)];
              if (cursor < removed_left) {
                  add_interval(1, 1, universe_count, cursor,
                               removed_left - 1, id, hulls);
              }
              cursor = max(cursor, removed_right + 1);
          }
          if (cursor <= right) {
              add_interval(1, 1, universe_count, cursor, right, id, hulls);
          }
      }
      build_hulls(hulls, planets);

      vector<Query> queries;
      queries.reserve(static_cast<size_t>(query_count));
      for (int id = 0; id < query_count; ++id) {
          int universe;
          long long x;
          cin >> universe >> x;
          queries.push_back({universe, x, id});
      }
      sort(queries.begin(), queries.end(), [](const Query& first,
                                               const Query& second) {
          if (first.x != second.x) { return first.x < second.x; }
          return first.id < second.id;
      });

      vector<size_t> pointer(hulls.size());
      vector<long long> answer(static_cast<size_t>(query_count));
      for (const Query& query : queries) {
          answer[static_cast<size_t>(query.id)] = query_path(
              tin[static_cast<size_t>(query.universe)], query.x,
              universe_count, hulls, pointer, planets);
      }
      for (const long long result : answer) {
          cout << result << '\n';
      }
      return 0;
  }
