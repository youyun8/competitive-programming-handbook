cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int tests; cin >> tests;
      while (tests--) {
          int n, friends; cin >> n >> friends;
          vector<double> area(static_cast<size_t>(n));
          double high = 0;
          for (double& x : area) { double r; cin >> r; x = acos(-1.0) * r * r; high = max(high, x); }
          const long long people = static_cast<long long>(friends) + 1;
          double low = 0;
          for (int iteration = 0; iteration < 100; ++iteration) {
              const double mid = (low + high) / 2;
              long long pieces = 0;
              for (const double x : area) {
                  pieces += static_cast<long long>(x / mid);
                  if (pieces >= people) break;
              }
              if (pieces >= people) low = mid; else high = mid;
          }
          cout << fixed << setprecision(4) << low << '\n';
      }
  }
