cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int n; double left, right; cin >> n >> left >> right;
      vector<double> coefficient(static_cast<size_t>(n) + 1U);
      for (double& value : coefficient) cin >> value;
      const auto evaluate = [&](double x) { double result = 0; for (const double value : coefficient) result = result * x + value; return result; };
      for (int iteration = 0; iteration < 150; ++iteration) {
          const double first = left + (right - left) / 3;
          const double second = right - (right - left) / 3;
          if (evaluate(first) < evaluate(second)) left = first; else right = second;
      }
      cout << fixed << setprecision(5) << (left + right) / 2 << '\n';
  }
