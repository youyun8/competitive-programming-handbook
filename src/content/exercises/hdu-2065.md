cpp_solution: |
  #include <iostream>
  using namespace std;
  int power_mod(int base, unsigned long long exponent) {
      int result = 1;
      while (exponent > 0U) {
          if ((exponent & 1U) != 0U) result = result * base % 100;
          base = base * base % 100;
          exponent >>= 1U;
      }
      return result;
  }
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int test_count;
      while (cin >> test_count && test_count != 0) {
          for (int case_index = 1; case_index <= test_count; ++case_index) {
              unsigned long long n;
              cin >> n;
              const int answer = (power_mod(4, n - 1U) + power_mod(2, n - 1U)) % 100;
              cout << "Case " << case_index << ": " << answer << '\n';
          }
          cout << '\n';
