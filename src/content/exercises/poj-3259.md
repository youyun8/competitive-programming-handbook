hints:
  - 普通道路加兩條權重 T 的邊；蟲洞只加 S→E、權重 -T。

  - Bellman-Ford 第 n 輪仍可鬆弛就必有負環。
solution_outline: 建立完整邊集，以全零距離執行 n 輪 Bellman-Ford；最後一輪有更新則輸出 YES。
proof_or_invariant: 全零初始化等價於超級源點以零權連向所有點。無負環時最短簡單路至多 n-1 條邊，第 n 輪不再更新；若仍更新，更新路徑含重複點且該環使距離下降，故為負環。

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  struct Edge{int from,to,weight;};
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int tests;if(!(cin>>tests))return 0;while(tests-->0){
          int n,m,w;cin>>n>>m>>w;vector<Edge> edges;
          for(int i=0;i<m;++i){int a,b,t;cin>>a>>b>>t;edges.push_back({a,b,t});edges.push_back({b,a,t});}
          for(int i=0;i<w;++i){int a,b,t;cin>>a>>b>>t;edges.push_back({a,b,-t});}
          vector<long long> dist(static_cast<size_t>(n+1),0);bool cycle=false;
          for(int round=1;round<=n;++round){bool changed=false;for(const Edge& e:edges)if(dist[static_cast<size_t>(e.to)]>dist[static_cast<size_t>(e.from)]+e.weight){dist[static_cast<size_t>(e.to)]=dist[static_cast<size_t>(e.from)]+e.weight;changed=true;if(round==n)cycle=true;}if(!changed)break;}
          cout<<(cycle?"YES":"NO")<<'\n';
      }
  }
