cpp_solution: |
  #include <iostream>
  #include <queue>
  #include <vector>
  using namespace std;
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);int n=0,m=0;
      while(cin>>n>>m&&(n!=0||m!=0)){
          vector<vector<int>> graph(static_cast<size_t>(n));vector<int> indegree(static_cast<size_t>(n),0);
          for(int i=0;i<m;++i){int u=0,v=0;cin>>u>>v;graph[static_cast<size_t>(u)].push_back(v);++indegree[static_cast<size_t>(v)];}
          queue<int> ready;for(int v=0;v<n;++v)if(indegree[static_cast<size_t>(v)]==0)ready.push(v);int processed=0;
          while(!ready.empty()){int u=ready.front();ready.pop();++processed;for(int v:graph[static_cast<size_t>(u)])if(--indegree[static_cast<size_t>(v)]==0)ready.push(v);}
          cout<<(processed==n?"YES\n":"NO\n");
      }
  }
