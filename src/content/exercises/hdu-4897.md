hints:
  - 為操作 2 想像把路徑每個頂點的標記 XOR 1；一條邊兩端標記 XOR 改變，恰好等價於只有一端在路徑。
  - 對輕邊可在查詢時計算兩端頂點標記 XOR；同一重鏈的內部重邊兩端一起翻，不變。
  - 每個重鏈頂點區間只有至多兩條重邊跨越區間邊界；操作 2 額外把這兩條重邊的顯式顏色翻轉即可。
solution_outline: W 線段樹維護映射到子節點的顯式邊色，L 線段樹維護操作 2 的頂點標記。操作 1 是標準路徑邊翻轉。操作 2 拆成重鏈頂點區間，翻 L 區間並翻每段兩側跨界重邊。查詢重邊直接查 W，輕邊再 XOR 兩端 L。
proof_or_invariant: 對每條輕邊，實際顏色為 W xor 兩端 L；對重邊，區間內兩端 L 同步變化而抵消，跨區間邊界時則同步翻 W 補上影響。故兩種表示對每次操作都維持真實邊色；標準樹剖查詢累加路徑邊即得答案。

  #include <algorithm>
  #include <iostream>
  #include <utility>
  #include <vector>
  using namespace std;
  struct FlipTree {
      int n; vector<int> sum; vector<bool> lazy;
      explicit FlipTree(int size) : n(size), sum(static_cast<size_t>(4 * size)), lazy(static_cast<size_t>(4 * size)) {}
      void apply(int node, int left, int right) { sum[static_cast<size_t>(node)] = right - left + 1 - sum[static_cast<size_t>(node)]; lazy[static_cast<size_t>(node)] = !lazy[static_cast<size_t>(node)]; }
      void push(int node, int left, int right) { if (!lazy[static_cast<size_t>(node)] || left == right) return; int middle = (left + right) / 2; apply(node * 2, left, middle); apply(node * 2 + 1, middle + 1, right); lazy[static_cast<size_t>(node)] = false; }
      void flip(int node, int left, int right, int query_left, int query_right) { if (query_right < left || right < query_left) return; if (query_left <= left && right <= query_right) { apply(node, left, right); return; } push(node, left, right); int middle = (left + right) / 2; flip(node * 2, left, middle, query_left, query_right); flip(node * 2 + 1, middle + 1, right, query_left, query_right); sum[static_cast<size_t>(node)] = sum[static_cast<size_t>(node * 2)] + sum[static_cast<size_t>(node * 2 + 1)]; }
      int query(int node, int left, int right, int query_left, int query_right) { if (query_right < left || right < query_left) return 0; if (query_left <= left && right <= query_right) return sum[static_cast<size_t>(node)]; push(node, left, right); int middle = (left + right) / 2; return query(node * 2, left, middle, query_left, query_right) + query(node * 2 + 1, middle + 1, right, query_left, query_right); }
      void flip(int left, int right) { if (left <= right) flip(1, 1, n, left, right); }
      int query(int left, int right) { return left <= right ? query(1, 1, n, left, right) : 0; }
  };
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int test_count; cin >> test_count;
      while (test_count--) {
          int n; cin >> n; vector<vector<int>> graph(static_cast<size_t>(n) + 1U);
          for (int i = 1; i < n; ++i) { int u, v; cin >> u >> v; graph[static_cast<size_t>(u)].push_back(v); graph[static_cast<size_t>(v)].push_back(u); }
          vector<int> parent(static_cast<size_t>(n) + 1U), depth(static_cast<size_t>(n) + 1U), order{1}; depth[1] = 1;
          for (size_t i = 0; i < order.size(); ++i) for (int next : graph[static_cast<size_t>(order[i])]) if (next != parent[static_cast<size_t>(order[i])]) { parent[static_cast<size_t>(next)] = order[i]; depth[static_cast<size_t>(next)] = depth[static_cast<size_t>(order[i])] + 1; order.push_back(next); }
          vector<int> size(static_cast<size_t>(n) + 1U, 1), heavy(static_cast<size_t>(n) + 1U);
          for (size_t i = order.size(); i-- > 0;) for (int next : graph[static_cast<size_t>(order[i])]) if (parent[static_cast<size_t>(next)] == order[i]) { size[static_cast<size_t>(order[i])] += size[static_cast<size_t>(next)]; if (size[static_cast<size_t>(next)] > size[static_cast<size_t>(heavy[static_cast<size_t>(order[i])])]) heavy[static_cast<size_t>(order[i])] = next; }
          vector<int> head(static_cast<size_t>(n) + 1U), position(static_cast<size_t>(n) + 1U), chain_end(static_cast<size_t>(n) + 1U); vector<pair<int, int>> stack{{1, 1}}; int timer = 0;
          while (!stack.empty()) { auto [start, chain_head] = stack.back(); stack.pop_back(); for (int node = start; node != 0; node = heavy[static_cast<size_t>(node)]) { head[static_cast<size_t>(node)] = chain_head; position[static_cast<size_t>(node)] = ++timer; chain_end[static_cast<size_t>(chain_head)] = timer; for (int next : graph[static_cast<size_t>(node)]) if (parent[static_cast<size_t>(next)] == node && next != heavy[static_cast<size_t>(node)]) stack.push_back({next, next}); } }
          FlipTree edge_color(n), vertex_mark(n);
          auto flip_path_edges = [&](int x, int y) { while (head[static_cast<size_t>(x)] != head[static_cast<size_t>(y)]) { if (depth[static_cast<size_t>(head[static_cast<size_t>(x)])] < depth[static_cast<size_t>(head[static_cast<size_t>(y)])]) swap(x, y); edge_color.flip(position[static_cast<size_t>(head[static_cast<size_t>(x)])], position[static_cast<size_t>(x)]); x = parent[static_cast<size_t>(head[static_cast<size_t>(x)])]; } if (depth[static_cast<size_t>(x)] > depth[static_cast<size_t>(y)]) swap(x, y); edge_color.flip(position[static_cast<size_t>(x)] + 1, position[static_cast<size_t>(y)]); };
          auto process_vertex_interval = [&](int chain_head, int left, int right) { vertex_mark.flip(left, right); if (left > position[static_cast<size_t>(chain_head)]) edge_color.flip(left, left); if (right < chain_end[static_cast<size_t>(chain_head)]) edge_color.flip(right + 1, right + 1); };
          auto flip_adjacent = [&](int x, int y) { while (head[static_cast<size_t>(x)] != head[static_cast<size_t>(y)]) { if (depth[static_cast<size_t>(head[static_cast<size_t>(x)])] < depth[static_cast<size_t>(head[static_cast<size_t>(y)])]) swap(x, y); int chain_head = head[static_cast<size_t>(x)]; process_vertex_interval(chain_head, position[static_cast<size_t>(chain_head)], position[static_cast<size_t>(x)]); x = parent[static_cast<size_t>(chain_head)]; } if (depth[static_cast<size_t>(x)] > depth[static_cast<size_t>(y)]) swap(x, y); process_vertex_interval(head[static_cast<size_t>(x)], position[static_cast<size_t>(x)], position[static_cast<size_t>(y)]); };
          auto query_path = [&](int x, int y) { int answer = 0; while (head[static_cast<size_t>(x)] != head[static_cast<size_t>(y)]) { if (depth[static_cast<size_t>(head[static_cast<size_t>(x)])] < depth[static_cast<size_t>(head[static_cast<size_t>(y)])]) swap(x, y); int chain_head = head[static_cast<size_t>(x)]; answer += edge_color.query(position[static_cast<size_t>(chain_head)] + 1, position[static_cast<size_t>(x)]); int light = edge_color.query(position[static_cast<size_t>(chain_head)], position[static_cast<size_t>(chain_head)]) ^ vertex_mark.query(position[static_cast<size_t>(chain_head)], position[static_cast<size_t>(chain_head)]) ^ vertex_mark.query(position[static_cast<size_t>(parent[static_cast<size_t>(chain_head)])], position[static_cast<size_t>(parent[static_cast<size_t>(chain_head)])]); answer += light; x = parent[static_cast<size_t>(chain_head)]; } if (depth[static_cast<size_t>(x)] > depth[static_cast<size_t>(y)]) swap(x, y); answer += edge_color.query(position[static_cast<size_t>(x)] + 1, position[static_cast<size_t>(y)]); return answer; };
          int q; cin >> q;
          while (q--) { int type, x, y; cin >> type >> x >> y; if (type == 1) flip_path_edges(x, y); else if (type == 2) flip_adjacent(x, y); else cout << query_path(x, y) << '\n'; }
      }
      return 0;
  }
