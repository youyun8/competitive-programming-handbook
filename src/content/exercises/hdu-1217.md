cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int n,case_no=0;while(cin>>n&&n!=0){
          unordered_map<string,int> id;string name;
          for(int i=0;i<n;++i){cin>>name;id[name]=i;}
          vector<vector<double>> best(static_cast<size_t>(n),vector<double>(static_cast<size_t>(n),0));
          for(int i=0;i<n;++i)best[static_cast<size_t>(i)][static_cast<size_t>(i)]=1;
          int m;cin>>m;for(int i=0;i<m;++i){string a,b;double rate;cin>>a>>rate>>b;best[static_cast<size_t>(id[a])][static_cast<size_t>(id[b])]=max(best[static_cast<size_t>(id[a])][static_cast<size_t>(id[b])],rate);}
          for(int k=0;k<n;++k)for(int i=0;i<n;++i)for(int j=0;j<n;++j)best[static_cast<size_t>(i)][static_cast<size_t>(j)]=max(best[static_cast<size_t>(i)][static_cast<size_t>(j)],best[static_cast<size_t>(i)][static_cast<size_t>(k)]*best[static_cast<size_t>(k)][static_cast<size_t>(j)]);
          bool yes=false;for(int i=0;i<n;++i)if(best[static_cast<size_t>(i)][static_cast<size_t>(i)]>1.0+1e-12)yes=true;
          cout<<"Case "<<++case_no<<": "<<(yes?"Yes":"No")<<'\n';
      }
  }
