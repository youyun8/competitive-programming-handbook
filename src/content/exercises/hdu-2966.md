  struct Point {
      long long coordinate[2]{};
      int id = 0;
  };
  struct Node {
      Point point;
      int child[2]{};
      long long minimum[2]{}, maximum[2]{};
  };
  vector<Point> points;
  vector<Node> tree;

  void pull(int node) {
      for (int dimension = 0; dimension < 2; ++dimension) {
          tree[node].minimum[dimension] = tree[node].maximum[dimension] =
              tree[node].point.coordinate[dimension];
          for (int side = 0; side < 2; ++side) {
              int next = tree[node].child[side];
              if (next == 0) continue;
              tree[node].minimum[dimension] = min(tree[node].minimum[dimension], tree[next].minimum[dimension]);
              tree[node].maximum[dimension] = max(tree[node].maximum[dimension], tree[next].maximum[dimension]);
          }
      }
  }
  int build(int left, int right, int dimension) {
      if (left >= right) return 0;
      int middle = (left + right) / 2;
      nth_element(points.begin() + left, points.begin() + middle, points.begin() + right,
                  [dimension](const Point& first, const Point& second) {
                      return first.coordinate[dimension] < second.coordinate[dimension];
                  });
      int node = middle + 1;
      tree[node].point = points[middle];
      tree[node].child[0] = build(left, middle, dimension ^ 1);
      tree[node].child[1] = build(middle + 1, right, dimension ^ 1);
      pull(node);
      return node;
  }
  long long square(long long value) { return value * value; }
  long long distance(const Point& first, const Point& second) {
      return square(first.coordinate[0] - second.coordinate[0]) +
             square(first.coordinate[1] - second.coordinate[1]);
  }
  long long box_distance(int node, const Point& query) {
      long long result = 0;
      for (int dimension = 0; dimension < 2; ++dimension) {
          if (query.coordinate[dimension] < tree[node].minimum[dimension])
              result += square(tree[node].minimum[dimension] - query.coordinate[dimension]);
          else if (query.coordinate[dimension] > tree[node].maximum[dimension])
              result += square(query.coordinate[dimension] - tree[node].maximum[dimension]);
      }
      return result;
  }
  void query(int node, const Point& target, long long& answer) {
      if (node == 0) return;
      if (tree[node].point.id != target.id) answer = min(answer, distance(tree[node].point, target));
      int first = tree[node].child[0], second = tree[node].child[1];
      long long first_bound = first == 0 ? numeric_limits<long long>::max() : box_distance(first, target);
      long long second_bound = second == 0 ? numeric_limits<long long>::max() : box_distance(second, target);
      if (first_bound > second_bound) {
          swap(first, second);
          swap(first_bound, second_bound);
      }
      if (first != 0 && first_bound < answer) query(first, target, answer);
      if (second != 0 && second_bound < answer) query(second, target, answer);
  }
