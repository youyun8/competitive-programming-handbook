cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  using Edge = pair<int, long long>;
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int n, m; long long health; cin >> n >> m >> health;
      vector<long long> fee(static_cast<size_t>(n)), candidates;
      for (auto& x : fee) { cin >> x; candidates.push_back(x); }
      vector<vector<Edge>> graph(static_cast<size_t>(n));
      for (int i = 0; i < m; ++i) {
          int u, v; long long w; cin >> u >> v >> w; --u; --v;
          graph[static_cast<size_t>(u)].push_back({v, w});
          graph[static_cast<size_t>(v)].push_back({u, w});
      }
      sort(candidates.begin(), candidates.end());
      candidates.erase(unique(candidates.begin(), candidates.end()), candidates.end());
      const auto feasible = [&](long long limit) {
          if (fee[0] > limit || fee.back() > limit) return false;
          const long long inf = numeric_limits<long long>::max() / 4;
          vector<long long> dist(static_cast<size_t>(n), inf);
          priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
          dist[0] = 0; pq.push({0, 0});
          while (!pq.empty()) {
              const auto [distance, u] = pq.top(); pq.pop();
              if (distance != dist[static_cast<size_t>(u)]) continue;
              for (const auto& [v, weight] : graph[static_cast<size_t>(u)]) {
                  if (fee[static_cast<size_t>(v)] > limit) continue;
                  const long long next = distance + weight;
                  if (next < dist[static_cast<size_t>(v)] && next <= health) {
                      dist[static_cast<size_t>(v)] = next; pq.push({next, v});
                  }
              }
          }
          return dist.back() <= health;
      };
      if (!feasible(candidates.back())) { cout << "AFK\n"; return 0; }
      size_t low = 0, high = candidates.size() - 1;
      while (low < high) {
          const size_t mid = low + (high - low) / 2;
          if (feasible(candidates[mid])) high = mid; else low = mid + 1;
      }
      cout << candidates[low] << '\n'; return 0;
  }
