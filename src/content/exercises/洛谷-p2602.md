cpp_solution: |
  #include <array>
  #include <iostream>
  using namespace std;
  static array<long long, 10> count_to(long long x) {
      array<long long, 10> result{};
      if (x <= 0) return result;
      for (long long factor = 1; factor <= x; ) {
          const long long lower = x % factor;
          const long long current = (x / factor) % 10;
          const long long higher = x / factor / 10;
          for (int digit = 1; digit <= 9; ++digit) {
              result[static_cast<size_t>(digit)] += higher * factor;
              if (current > digit) result[static_cast<size_t>(digit)] += factor;
              else if (current == digit) result[static_cast<size_t>(digit)] += lower + 1;
          }
          if (higher > 0) {
              result[0] += (higher - 1) * factor;
              result[0] += current == 0 ? lower + 1 : factor;
          }
          if (factor > x / 10) break;
          factor *= 10;
      }
      return result;
  }
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      long long a = 0, b = 0; cin >> a >> b;
      const auto right = count_to(b);
      const auto left = count_to(a - 1);
      for (size_t digit = 0; digit < 10; ++digit) {
          if (digit > 0) cout << ' ';
          cout << right[digit] - left[digit];
      }
      cout << '\n';
  }
