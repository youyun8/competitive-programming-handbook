cpp_solution: |-
  #if defined(__GNUC__)
  #pragma GCC diagnostic ignored "-Wconversion"
  #pragma GCC diagnostic ignored "-Wshadow"
  #pragma GCC diagnostic ignored "-Wpedantic"
  #pragma GCC diagnostic ignored "-Wsign-compare"
  #pragma GCC diagnostic ignored "-Wunused-parameter"
  #pragma GCC diagnostic ignored "-Wunused-variable"
  #pragma GCC diagnostic ignored "-Wunused-function"
  #pragma GCC diagnostic ignored "-Wunused-result"
  #pragma GCC diagnostic ignored "-Wparentheses"
  #pragma GCC diagnostic ignored "-Wmisleading-indentation"

  #pragma GCC diagnostic ignored "-Wclass-memaccess"
  #pragma GCC diagnostic ignored "-Wimplicit-fallthrough"

  #include <algorithm>
  #include <iostream>
  #include <limits>
  #include <vector>
  using namespace std;

  struct Hotel {
      int x = 0, y = 0, price = 0, id = 0;
  };
  struct Node {
      Hotel hotel;
      int child[2]{};
      int minimum_x = 0, maximum_x = 0, minimum_y = 0, maximum_y = 0;
      int minimum_price = 0;
  };
  vector<Hotel> hotels;
  vector<Node> tree;

  void pull(int node) {
      Node& current = tree[node];
      current.minimum_x = current.maximum_x = current.hotel.x;
      current.minimum_y = current.maximum_y = current.hotel.y;
      current.minimum_price = current.hotel.price;
      for (int side = 0; side < 2; ++side) {
          int next = current.child[side];
          if (next == 0) continue;
          current.minimum_x = min(current.minimum_x, tree[next].minimum_x);
          current.maximum_x = max(current.maximum_x, tree[next].maximum_x);
          current.minimum_y = min(current.minimum_y, tree[next].minimum_y);
          current.maximum_y = max(current.maximum_y, tree[next].maximum_y);
          current.minimum_price = min(current.minimum_price, tree[next].minimum_price);
      }
  }
  int build(int left, int right, int dimension) {
      if (left >= right) return 0;
      int middle = (left + right) / 2;
      nth_element(hotels.begin() + left, hotels.begin() + middle, hotels.begin() + right,
                  [dimension](const Hotel& first, const Hotel& second) {
                      return dimension == 0 ? first.x < second.x : first.y < second.y;
                  });
      int node = middle + 1;
      tree[node].hotel = hotels[static_cast<size_t>(middle)];
      tree[node].child[0] = build(left, middle, dimension ^ 1);
      tree[node].child[1] = build(middle + 1, right, dimension ^ 1);
      pull(node);
      return node;
  }
  long long square(long long value) { return value * value; }
  long long point_distance(const Hotel& hotel, int x, int y) {
      return square(static_cast<long long>(hotel.x) - x) +
             square(static_cast<long long>(hotel.y) - y);
  }
  long long box_distance(int node, int x, int y) {
      long long answer = 0;
      if (x < tree[node].minimum_x) answer += square(static_cast<long long>(tree[node].minimum_x) - x);
      if (x > tree[node].maximum_x) answer += square(static_cast<long long>(x) - tree[node].maximum_x);
      if (y < tree[node].minimum_y) answer += square(static_cast<long long>(tree[node].minimum_y) - y);
      if (y > tree[node].maximum_y) answer += square(static_cast<long long>(y) - tree[node].maximum_y);
      return answer;
  }
  void search(int node, int x, int y, int maximum_price, long long& best_distance, int& best_id,
              Hotel& answer) {
      if (node == 0 || tree[node].minimum_price > maximum_price) return;
      const Hotel& hotel = tree[node].hotel;
      if (hotel.price <= maximum_price) {
          long long candidate = point_distance(hotel, x, y);
          if (candidate < best_distance || (candidate == best_distance && hotel.id < best_id)) {
              best_distance = candidate;
              best_id = hotel.id;
              answer = hotel;
          }
      }
      int first = tree[node].child[0], second = tree[node].child[1];
      long long first_bound = first == 0 ? numeric_limits<long long>::max() : box_distance(first, x, y);
      long long second_bound = second == 0 ? numeric_limits<long long>::max() : box_distance(second, x, y);
      if (first_bound > second_bound) {
          swap(first, second);
          swap(first_bound, second_bound);
      }
      if (first != 0 && first_bound <= best_distance)
          search(first, x, y, maximum_price, best_distance, best_id, answer);
      if (second != 0 && second_bound <= best_distance)
          search(second, x, y, maximum_price, best_distance, best_id, answer);
  }
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int test_count = 0;
      cin >> test_count;
      while (test_count-- > 0) {
          int n = 0, query_count = 0;
          cin >> n >> query_count;
          hotels.assign(static_cast<size_t>(n), {});
          for (int index = 0; index < n; ++index) {
              Hotel& hotel = hotels[static_cast<size_t>(index)];
              cin >> hotel.x >> hotel.y >> hotel.price;
              hotel.id = index;
          }
          tree.assign(static_cast<size_t>(n) + 1U, {});
          int root = build(0, n, 0);
          while (query_count-- > 0) {
              int x = 0, y = 0, maximum_price = 0;
              cin >> x >> y >> maximum_price;
              long long best_distance = numeric_limits<long long>::max();
              int best_id = numeric_limits<int>::max();
              Hotel answer;
              search(root, x, y, maximum_price, best_distance, best_id, answer);
              cout << answer.x << ' ' << answer.y << ' ' << answer.price << '\n';
          }
      }
      return 0;
  }
