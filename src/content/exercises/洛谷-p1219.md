cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  static int n;
  static long long solution_count = 0;
  static vector<int> placement;
  static vector<bool> used_column, used_down, used_up;
  static void search(int row) {
      if (row == n) {
          ++solution_count;
          if (solution_count <= 3) {
              for (int column = 0; column < n; ++column) { if (column > 0) cout << ' '; cout << placement[column] + 1; }
              cout << '\n';
          }
          return;
      }
      for (int column = 0; column < n; ++column) {
          const int down = row - column + n - 1;
          const int up = row + column;
          if (used_column[column] || used_down[down] || used_up[up]) continue;
          placement[row] = column;
          used_column[column] = used_down[down] = used_up[up] = true;
          search(row + 1);
          used_column[column] = used_down[down] = used_up[up] = false;
      }
  }
  int main() {
      cin >> n;
      placement.resize(n);
      used_column.assign(n, false);
      used_down.assign(2 * n - 1, false);
      used_up.assign(2 * n - 1, false);
      search(0);
      cout << solution_count << '\n';
  }
