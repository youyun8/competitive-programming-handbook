  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int tests;if(!(cin>>tests))return 0;for(int case_no=1;case_no<=tests;++case_no){int n,m;cin>>n>>m;vector<array<int,2>> choice(static_cast<size_t>(n));for(auto& value:choice){cin>>value[0];value[1]=value[0]%3+1;}vector<vector<int>> graph(static_cast<size_t>(2*n));auto forbid=[&](int i,int x,int j,int y){graph[static_cast<size_t>(2*i+x)].push_back(2*j+(y^1));graph[static_cast<size_t>(2*j+y)].push_back(2*i+(x^1));};
          for(int z=0;z<m;++z){int a,b,k;cin>>a>>b>>k;--a;--b;for(int x=0;x<2;++x)for(int y=0;y<2;++y){bool same=choice[static_cast<size_t>(a)][static_cast<size_t>(x)]==choice[static_cast<size_t>(b)][static_cast<size_t>(y)];if(same!=(k==0))forbid(a,x,b,y);}}
          vector<int> dfn(static_cast<size_t>(2*n)),low(static_cast<size_t>(2*n)),component(static_cast<size_t>(2*n),-1),stack_nodes;vector<char> on_stack(static_cast<size_t>(2*n));int timer=0,count=0;auto dfs=[&](auto&& self,int u)->void{dfn[static_cast<size_t>(u)]=low[static_cast<size_t>(u)]=++timer;stack_nodes.push_back(u);on_stack[static_cast<size_t>(u)]=1;for(int v:graph[static_cast<size_t>(u)])if(!dfn[static_cast<size_t>(v)]){self(self,v);low[static_cast<size_t>(u)]=min(low[static_cast<size_t>(u)],low[static_cast<size_t>(v)]);}else if(on_stack[static_cast<size_t>(v)])low[static_cast<size_t>(u)]=min(low[static_cast<size_t>(u)],dfn[static_cast<size_t>(v)]);if(low[static_cast<size_t>(u)]==dfn[static_cast<size_t>(u)]){while(true){int v=stack_nodes.back();stack_nodes.pop_back();on_stack[static_cast<size_t>(v)]=0;component[static_cast<size_t>(v)]=count;if(v==u)break;}++count;}};
