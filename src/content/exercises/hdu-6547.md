judgment: 開根號使用向下取整；0 與 1 再開根號不變，可整段跳過。
hints:
  - 樹路徑先以重鏈剖分拆成 O(log n) 個 DFS 序區間。
  - 對一般區間無法用懶標記代表開根號，但若區間最大值 <=1，整段永遠不再改變。
  - 線段樹同時維護 sum 與 maximum；更新只遞迴進入與詢問相交且 maximum>1 的節點。
solution_outline: 樹剖後用線段樹維護 DFS 序。每段路徑更新執行帶 maximum 剪枝的區間 sqrt；查詢則累加每段區間和。
proof_or_invariant: 葉節點更新精確套用 floor(sqrt)。被剪枝區間所有值皆為 0 或 1，更新前後相同，故略過安全。樹剖區間完整分割路徑，因此更新與查詢都與原樹操作等價。
common_errors: [以區間和判斷是否可剪枝, 浮點 sqrt 後未向下取整, 路徑查詢漏掉同鏈最後一段]

  #include <algorithm>
  #include <cmath>
  #include <iostream>
  #include <utility>
  #include <vector>
  using namespace std;
  struct SegmentTree {
      int n; vector<long long> sum, maximum; const vector<long long>& base;
      SegmentTree(int size, const vector<long long>& values)
          : n(size), sum(static_cast<size_t>(4 * size)), maximum(static_cast<size_t>(4 * size)), base(values) { build(1, 1, n); }
      void pull(int node) { sum[static_cast<size_t>(node)] = sum[static_cast<size_t>(node * 2)] + sum[static_cast<size_t>(node * 2 + 1)]; maximum[static_cast<size_t>(node)] = max(maximum[static_cast<size_t>(node * 2)], maximum[static_cast<size_t>(node * 2 + 1)]); }
      void build(int node, int left, int right) { if (left == right) { sum[static_cast<size_t>(node)] = maximum[static_cast<size_t>(node)] = base[static_cast<size_t>(left)]; return; } int middle = (left + right) / 2; build(node * 2, left, middle); build(node * 2 + 1, middle + 1, right); pull(node); }
      void update(int node, int left, int right, int query_left, int query_right) {
          if (query_right < left || right < query_left || maximum[static_cast<size_t>(node)] <= 1) return;
          if (left == right) { const long long value = static_cast<long long>(sqrt(static_cast<long double>(sum[static_cast<size_t>(node)]))); sum[static_cast<size_t>(node)] = maximum[static_cast<size_t>(node)] = value; return; }
          int middle = (left + right) / 2; update(node * 2, left, middle, query_left, query_right); update(node * 2 + 1, middle + 1, right, query_left, query_right); pull(node);
      }
      long long query(int node, int left, int right, int query_left, int query_right) const {
          if (query_right < left || right < query_left) return 0;
          if (query_left <= left && right <= query_right) return sum[static_cast<size_t>(node)];
          int middle = (left + right) / 2; return query(node * 2, left, middle, query_left, query_right) + query(node * 2 + 1, middle + 1, right, query_left, query_right);
      }
      void update(int left, int right) { update(1, 1, n, left, right); }
      long long query(int left, int right) const { return query(1, 1, n, left, right); }
  };
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int n, q; cin >> n >> q;
      vector<long long> value(static_cast<size_t>(n) + 1U); for (int i = 1; i <= n; ++i) cin >> value[static_cast<size_t>(i)];
      vector<vector<int>> graph(static_cast<size_t>(n) + 1U);
      for (int i = 1; i < n; ++i) { int u, v; cin >> u >> v; graph[static_cast<size_t>(u)].push_back(v); graph[static_cast<size_t>(v)].push_back(u); }
      vector<int> parent(static_cast<size_t>(n) + 1U), depth(static_cast<size_t>(n) + 1U), order{1}; depth[1] = 1;
      for (size_t i = 0; i < order.size(); ++i) for (int next : graph[static_cast<size_t>(order[i])]) if (next != parent[static_cast<size_t>(order[i])]) { parent[static_cast<size_t>(next)] = order[i]; depth[static_cast<size_t>(next)] = depth[static_cast<size_t>(order[i])] + 1; order.push_back(next); }
      vector<int> size(static_cast<size_t>(n) + 1U, 1), heavy(static_cast<size_t>(n) + 1U);
      for (size_t i = order.size(); i-- > 0;) for (int next : graph[static_cast<size_t>(order[i])]) if (parent[static_cast<size_t>(next)] == order[i]) { size[static_cast<size_t>(order[i])] += size[static_cast<size_t>(next)]; if (size[static_cast<size_t>(next)] > size[static_cast<size_t>(heavy[static_cast<size_t>(order[i])])]) heavy[static_cast<size_t>(order[i])] = next; }
      vector<int> head(static_cast<size_t>(n) + 1U), position(static_cast<size_t>(n) + 1U); vector<long long> base(static_cast<size_t>(n) + 1U);
      vector<pair<int, int>> stack{{1, 1}}; int timer = 0;
      while (!stack.empty()) { auto [start, chain_head] = stack.back(); stack.pop_back(); for (int node = start; node != 0; node = heavy[static_cast<size_t>(node)]) { head[static_cast<size_t>(node)] = chain_head; position[static_cast<size_t>(node)] = ++timer; base[static_cast<size_t>(timer)] = value[static_cast<size_t>(node)]; for (int next : graph[static_cast<size_t>(node)]) if (parent[static_cast<size_t>(next)] == node && next != heavy[static_cast<size_t>(node)]) stack.push_back({next, next}); } }
      SegmentTree tree(n, base);
      while (q--) {
          int operation, x, y; cin >> operation >> x >> y;
          long long answer = 0;
          while (head[static_cast<size_t>(x)] != head[static_cast<size_t>(y)]) {
              if (depth[static_cast<size_t>(head[static_cast<size_t>(x)])] < depth[static_cast<size_t>(head[static_cast<size_t>(y)])]) swap(x, y);
              int left = position[static_cast<size_t>(head[static_cast<size_t>(x)])], right = position[static_cast<size_t>(x)];
              if (operation == 0) tree.update(left, right); else answer += tree.query(left, right);
              x = parent[static_cast<size_t>(head[static_cast<size_t>(x)])];
          }
          if (position[static_cast<size_t>(x)] > position[static_cast<size_t>(y)]) swap(x, y);
          if (operation == 0) tree.update(position[static_cast<size_t>(x)], position[static_cast<size_t>(y)]);
          else { answer += tree.query(position[static_cast<size_t>(x)], position[static_cast<size_t>(y)]); cout << answer << '\n'; }
      }
      return 0;
  }
