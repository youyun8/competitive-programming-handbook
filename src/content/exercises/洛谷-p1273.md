solution_outline: >-
  後序 DFS。葉節點初始化 0 人收益 0、1 人收益付款；內部節點逐一用分組揹包合併兒子，取正數客戶時扣該邊費用。根陣列中最大非負容量即答案。
proof_or_invariant: >-

cpp_solution: |
  #include <algorithm>
  #include <iostream>
  #include <limits>
  #include <utility>
  #include <vector>
  using namespace std;
  struct Edge { int to; int cost; };
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int n, m;
      cin >> n >> m;
      const int internal = n - m;
      vector<vector<Edge>> children(static_cast<size_t>(n + 1));
      for (int u = 1; u <= internal; ++u) {
          int count; cin >> count;
          while (count-- > 0) {
              int v, cost; cin >> v >> cost;
              children[u].push_back({v, cost});
          }
      }
      vector<int> payment(static_cast<size_t>(n + 1));
      for (int u = internal + 1; u <= n; ++u) cin >> payment[u];
      const int negative = numeric_limits<int>::min() / 4;
      vector<vector<int>> dp(static_cast<size_t>(n + 1),
                             vector<int>(static_cast<size_t>(m + 1), negative));
      vector<int> leaves(static_cast<size_t>(n + 1));
      auto dfs = [&](auto&& self, int u) -> void {
          dp[u][0] = 0;
          if (u > internal) {
              dp[u][1] = payment[u];
              leaves[u] = 1;
              return;
          }
          for (Edge edge : children[u]) {
              self(self, edge.to);
              for (int total = min(m, leaves[u] + leaves[edge.to]); total >= 0; --total) {
                  for (int take = 1; take <= min(total, leaves[edge.to]); ++take) {
                      if (dp[u][total - take] == negative || dp[edge.to][take] == negative) continue;
                      dp[u][total] = max(dp[u][total],
                          dp[u][total - take] + dp[edge.to][take] - edge.cost);
                  }
              }
              leaves[u] += leaves[edge.to];
          }
      };
      dfs(dfs, 1);
      for (int answer = m; answer >= 0; --answer)
          if (dp[1][answer] >= 0) { cout << answer << '\n'; break; }
  }
