statement: 連通無向圖中，每個等機率方案會把指定道路成本提高（各方案互不影響）。求每次改價後最小生成樹成本的平均值。
constraints: [測試組數不超過 20, 1 <= n <= 3000, m <= n*n, q <= 10000, 單邊新舊成本 <= 10^7, 無重邊, 原圖連通]
input_format: 多組 n、m（0 0 結束）；m 行 0-based 的 x、y、c；接著 q 與 q 行 x、y、新成本。
output_format: 每組輸出平均 MST 成本，固定四位小數。

      0 1 6

      6.0000
    explanation: 官方範例三次成本為 6、5、7，平均 6；另以小圖逐次重跑 Kruskal 對拍替代邊表。
core_knowledge: [MST 敏感度, 每條樹邊最小替代邊, O(n²) 樹形 DP]
judgment: 非 MST 邊漲價不影響既有 MST；MST 邊漲價時只需比較新成本與跨該切割的最輕替代邊。
hints:
  - 先求任一 MST，記總成本與哪些原邊被選。
  - 對每條樹邊，預處理刪除後兩分量之間最輕的非樹邊。
  - 查詢樹邊 uv 時答案為 mst-w(uv)+min(new_w,replacement[uv])；非樹邊仍為 mst。
solution_outline: Kruskal 建 MST。以每個 root 為固定外部端點在 MST 上 DFS；子樹回傳 root 到該子樹任一點的最小非樹邊，並用回傳值更新父子樹邊的 replacement。總 O(n²)。逐詢問 O(1) 累加。
proof_or_invariant: 刪除樹邊 e 後，原 MST 其餘邊在兩側各自仍最優；任何新生成樹只需用一條跨割邊重接。切割中可用的最小成本即 replacement[e]。逐 root DFS 恰枚舉每條非樹邊的一端位於各子樹的情形，因此每條樹邊取得完整跨割最小值。
common_errors: [把各次漲價累積, 非樹邊查詢仍重算, replacement 誤包含原樹邊本身, 平均總和使用 int]
complexity: { time: 'O(m log m + n^2 + q)', space: 'O(n^2+m)' }

  #include <bits/stdc++.h>
  using namespace std;
  struct Edge{int u,v,w;};
  struct Dsu{vector<int>p;explicit Dsu(int n):p(static_cast<size_t>(n)){iota(p.begin(),p.end(),0);}int find(int x){return p[static_cast<size_t>(x)]==x?x:p[static_cast<size_t>(x)]=find(p[static_cast<size_t>(x)]);}bool unite(int x,int y){x=find(x);y=find(y);if(x==y)return false;p[static_cast<size_t>(x)]=y;return true;}};
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int n,m;cout<<fixed<<setprecision(4);
      while(cin>>n>>m&&(n||m)){
          vector<Edge> edges(static_cast<size_t>(m));
          const int infinity=INT_MAX/4;
          vector<vector<int>> cost(static_cast<size_t>(n),vector<int>(static_cast<size_t>(n),infinity));
          for(auto& edge:edges){cin>>edge.u>>edge.v>>edge.w;cost[static_cast<size_t>(edge.u)][static_cast<size_t>(edge.v)]=cost[static_cast<size_t>(edge.v)][static_cast<size_t>(edge.u)]=edge.w;}
          sort(edges.begin(),edges.end(),[](const Edge&a,const Edge&b){return a.w<b.w;});
          Dsu dsu(n);vector<vector<int>> tree(static_cast<size_t>(n));vector<vector<char>> in_tree(static_cast<size_t>(n),vector<char>(static_cast<size_t>(n)));long long mst=0;
          for(const auto& edge:edges)if(dsu.unite(edge.u,edge.v)){mst+=edge.w;tree[static_cast<size_t>(edge.u)].push_back(edge.v);tree[static_cast<size_t>(edge.v)].push_back(edge.u);in_tree[static_cast<size_t>(edge.u)][static_cast<size_t>(edge.v)]=in_tree[static_cast<size_t>(edge.v)][static_cast<size_t>(edge.u)]=1;}
          vector<vector<int>> replacement(static_cast<size_t>(n),vector<int>(static_cast<size_t>(n),infinity));
          function<int(int,int,int)> dfs=[&](int root,int node,int parent){int result=(root!=node&&!in_tree[static_cast<size_t>(root)][static_cast<size_t>(node)])?cost[static_cast<size_t>(root)][static_cast<size_t>(node)]:infinity;for(int next:tree[static_cast<size_t>(node)])if(next!=parent){int candidate=dfs(root,next,node);replacement[static_cast<size_t>(node)][static_cast<size_t>(next)]=replacement[static_cast<size_t>(next)][static_cast<size_t>(node)]=min(replacement[static_cast<size_t>(node)][static_cast<size_t>(next)],candidate);result=min(result,candidate);}return result;};
          for(int root=0;root<n;++root)dfs(root,root,-1);
          int query_count;cin>>query_count;long double total=0;
          for(int query=0;query<query_count;++query){int u,v,new_cost;cin>>u>>v>>new_cost;if(!in_tree[static_cast<size_t>(u)][static_cast<size_t>(v)])total+=mst;else total+=mst-cost[static_cast<size_t>(u)][static_cast<size_t>(v)]+min(new_cost,replacement[static_cast<size_t>(u)][static_cast<size_t>(v)]);}
          cout<<static_cast<double>(total/query_count)<<'\n';
      }
  }
