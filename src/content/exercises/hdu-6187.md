cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  struct Edge { int u, v, w; };
  class DisjointSet {
  public:
      explicit DisjointSet(int n) : parent_(static_cast<size_t>(n + 1)), size_(static_cast<size_t>(n + 1), 1) { iota(parent_.begin(), parent_.end(), 0); }
      int find_root(int x) { return parent_[static_cast<size_t>(x)] == x ? x : parent_[static_cast<size_t>(x)] = find_root(parent_[static_cast<size_t>(x)]); }
      bool unite(int x, int y) {
          x = find_root(x); y = find_root(y);
          if (x == y) return false;
          if (size_[static_cast<size_t>(x)] < size_[static_cast<size_t>(y)]) swap(x, y);
          parent_[static_cast<size_t>(y)] = x; size_[static_cast<size_t>(x)] += size_[static_cast<size_t>(y)]; return true;
      }
  private: vector<int> parent_, size_;
  };
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int n, m;
      while (cin >> n >> m) {
          for (int i = 0, x, y; i < n; ++i) cin >> x >> y;
          vector<Edge> edges(static_cast<size_t>(m));
          long long total_cost = 0;
          for (auto& edge : edges) { cin >> edge.u >> edge.v >> edge.w; total_cost += edge.w; }
          sort(edges.begin(), edges.end(), [](const Edge& a, const Edge& b) { return a.w > b.w; });
          DisjointSet dsu(n);
          int kept = 0;
          long long kept_cost = 0;
          for (const auto& edge : edges)
              if (dsu.unite(edge.u, edge.v)) { ++kept; kept_cost += edge.w; }
          cout << m - kept << ' ' << total_cost - kept_cost << '\n';
      }
  }
