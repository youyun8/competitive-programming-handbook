  #include <bits/stdc++.h>
  using namespace std;
  struct Dinic {
      struct Edge { int to; long long capacity; };
      vector<Edge> edges; vector<vector<int>> graph; vector<int> level; vector<size_t> current;
      explicit Dinic(int n) : graph(static_cast<size_t>(n)), level(static_cast<size_t>(n)), current(static_cast<size_t>(n)) {}
      void add_edge(int u, int v, long long c) {
          graph[static_cast<size_t>(u)].push_back(static_cast<int>(edges.size())); edges.push_back({v, c});
          graph[static_cast<size_t>(v)].push_back(static_cast<int>(edges.size())); edges.push_back({u, 0});
      }
      bool bfs(int source, int sink) {
          fill(level.begin(), level.end(), -1); queue<int> nodes; nodes.push(source); level[static_cast<size_t>(source)] = 0;
          while (!nodes.empty()) { int u = nodes.front(); nodes.pop(); for (int id : graph[static_cast<size_t>(u)]) { const Edge& e = edges[static_cast<size_t>(id)]; if (e.capacity > 0 && level[static_cast<size_t>(e.to)] < 0) { level[static_cast<size_t>(e.to)] = level[static_cast<size_t>(u)] + 1; nodes.push(e.to); } } }
          return level[static_cast<size_t>(sink)] >= 0;
      }
      long long dfs(int u, int sink, long long limit) {
          if (u == sink) return limit;
          for (size_t& i = current[static_cast<size_t>(u)]; i < graph[static_cast<size_t>(u)].size(); ++i) { int id = graph[static_cast<size_t>(u)][i]; Edge& e = edges[static_cast<size_t>(id)]; if (e.capacity <= 0 || level[static_cast<size_t>(e.to)] != level[static_cast<size_t>(u)] + 1) continue; long long sent = dfs(e.to, sink, min(limit, e.capacity)); if (sent > 0) { e.capacity -= sent; edges[static_cast<size_t>(id ^ 1)].capacity += sent; return sent; } }
          return 0;
      }
      long long flow(int source, int sink) {
          long long answer = 0;
          while (bfs(source, sink)) { fill(current.begin(), current.end(), 0); while (long long sent = dfs(source, sink, LLONG_MAX)) answer += sent; }
          return answer;
      }
  };
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int tests; cin >> tests;
      for (int case_id = 1; case_id <= tests; ++case_id) {
          int n, m; cin >> n >> m; Dinic dinic(n + 1);
          for (int i = 0, u, v, c; i < m; ++i) { cin >> u >> v >> c; dinic.add_edge(u, v, c); }
          cout << "Case " << case_id << ": " << dinic.flow(1, n) << '\n';
      }
  }
