hints:
  - 用 reachable[a][b] 表示已知 a 能勝過 b。
  - 布林 Floyd 的轉移是 reachable[i][j] |= reachable[i][k] && reachable[k][j]。
  - 最後只枚舉 i<j，兩個方向都不可達才計入答案。
solution_outline: 對直接勝負建有向可達矩陣，執行 Floyd 傳遞閉包，再統計互相皆不可達的無序點對。
proof_or_invariant: Floyd 第 k 輪後，矩陣恰包含所有只以 1..k 作中繼的勝負鏈。全部完成後，可達恰等價於可由傳遞性推出；雙向皆不可達的點對正是無法回答者。

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int tests;
      if (!(cin >> tests)) return 0;
      while (tests-- > 0) {
          int n, m; cin >> n >> m;
          vector<vector<char>> reachable(static_cast<size_t>(n), vector<char>(static_cast<size_t>(n), 0));
          for (int i = 0; i < m; ++i) {
              int a, b; cin >> a >> b;
              reachable[static_cast<size_t>(a - 1)][static_cast<size_t>(b - 1)] = 1;
          }
          for (int k = 0; k < n; ++k)
              for (int i = 0; i < n; ++i)
                  if (reachable[static_cast<size_t>(i)][static_cast<size_t>(k)])
                      for (int j = 0; j < n; ++j)
                          reachable[static_cast<size_t>(i)][static_cast<size_t>(j)] |=
                              reachable[static_cast<size_t>(k)][static_cast<size_t>(j)];
          int answer = 0;
          for (int i = 0; i < n; ++i)
              for (int j = i + 1; j < n; ++j)
                  if (!reachable[static_cast<size_t>(i)][static_cast<size_t>(j)] &&
                      !reachable[static_cast<size_t>(j)][static_cast<size_t>(i)]) ++answer;
          cout << answer << '\n';
      }
  }
