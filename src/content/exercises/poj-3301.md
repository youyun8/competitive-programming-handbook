solution_outline: evaluate(θ) 旋轉每點，求包圍盒寬高最大值。三分角度最小化邊長，平方為面積。
proof_or_invariant: 對固定方向，任何覆蓋正方形邊長至少為兩座標跨度最大值，該長度亦可構造，故 evaluate 精確。正方形方向以 π/2 週期重複，標準旋轉寬度函數在此基本區間可用三分定位最小值。

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  struct Point{double x,y;};
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int tests;cin>>tests;
      while(tests--){
          int n;cin>>n;vector<Point> points(static_cast<size_t>(n));for(auto&p:points)cin>>p.x>>p.y;
          const auto side=[&](double angle){
              double min_x=1e100,max_x=-1e100,min_y=1e100,max_y=-1e100;
              const double cosine=cos(angle),sine=sin(angle);
              for(const auto&p:points){double x=p.x*cosine-p.y*sine,y=p.x*sine+p.y*cosine;min_x=min(min_x,x);max_x=max(max_x,x);min_y=min(min_y,y);max_y=max(max_y,y);}
              return max(max_x-min_x,max_y-min_y);
          };
          double left=0,right=acos(-1.0)/2;
          for(int it=0;it<150;++it){double m1=left+(right-left)/3,m2=right-(right-left)/3;if(side(m1)<side(m2))right=m2;else left=m1;}
          const double length=side((left+right)/2);cout<<fixed<<setprecision(2)<<length*length<<'\n';
      }
  }
