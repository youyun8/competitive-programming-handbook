cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;

  struct Node { int left = 0, right = 0, count = 0; };

  static int insert_node(vector<Node>& tree, int previous, int low, int high, int position) {
      const int current = static_cast<int>(tree.size());
      tree.push_back(tree[static_cast<size_t>(previous)]);
      ++tree[static_cast<size_t>(current)].count;
      if (low == high) { return current; }
      const int middle = (low + high) / 2;
      if (position <= middle) {
          tree[static_cast<size_t>(current)].left =
              insert_node(tree, tree[static_cast<size_t>(previous)].left, low, middle, position);
      } else {
          tree[static_cast<size_t>(current)].right =
              insert_node(tree, tree[static_cast<size_t>(previous)].right, middle + 1, high, position);
      }
      return current;
  }

  static int prefix_count(const vector<Node>& tree, int older, int newer,
                          int low, int high, int bound) {
      if (bound < low) { return 0; }
      if (high <= bound) {
          return tree[static_cast<size_t>(newer)].count - tree[static_cast<size_t>(older)].count;
      }
      const int middle = (low + high) / 2;
      int answer = prefix_count(tree, tree[static_cast<size_t>(older)].left,
                                tree[static_cast<size_t>(newer)].left, low, middle, bound);
      if (bound > middle) {
          answer += prefix_count(tree, tree[static_cast<size_t>(older)].right,
                                 tree[static_cast<size_t>(newer)].right, middle + 1, high, bound);
      }
      return answer;
  }

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int test_count;
      cin >> test_count;
      for (int case_id = 1; case_id <= test_count; ++case_id) {
          int n, query_count;
          cin >> n >> query_count;
          vector<int> height(static_cast<size_t>(n));
          for (int& value : height) { cin >> value; }
          vector<int> values = height;
          sort(values.begin(), values.end());
          values.erase(unique(values.begin(), values.end()), values.end());
          vector<Node> tree(1);
          tree.reserve(static_cast<size_t>(n) * 20U + 1U);
          vector<int> root(static_cast<size_t>(n) + 1U);
          for (int i = 0; i < n; ++i) {
              const int position = static_cast<int>(
                  lower_bound(values.begin(), values.end(), height[static_cast<size_t>(i)]) - values.begin());
              root[static_cast<size_t>(i + 1)] =
                  insert_node(tree, root[static_cast<size_t>(i)], 0,
                              static_cast<int>(values.size()) - 1, position);
          }
          cout << "Case " << case_id << ":\n";
          while (query_count-- > 0) {
              int left, right, height_limit;
              cin >> left >> right >> height_limit;
              const int bound = static_cast<int>(
                  upper_bound(values.begin(), values.end(), height_limit) - values.begin()) - 1;
              cout << prefix_count(tree, root[static_cast<size_t>(left)],
                                   root[static_cast<size_t>(right + 1)], 0,
                                   static_cast<int>(values.size()) - 1, bound) << '\n';
          }
      }
  }
