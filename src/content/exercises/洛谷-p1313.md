cpp_solution: |
  #include <iostream>
  #include <vector>
  using namespace std;

  static constexpr long long mod_value = 10007;
  static long long power_mod(long long base, int exponent) {
      long long result = 1;
      base %= mod_value;
      while (exponent > 0) {
          if ((exponent & 1) != 0) { result = result * base % mod_value; }
          base = base * base % mod_value;
          exponent >>= 1;
      }
      return result;
  }

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      long long a, b;
      int k, n, m;
      cin >> a >> b >> k >> n >> m;
      vector<long long> combinations(static_cast<size_t>(k) + 1U);
      combinations[0] = 1;
      for (int i = 1; i <= k; ++i) {
          for (int j = i; j >= 1; --j) {
              combinations[static_cast<size_t>(j)] =
                  (combinations[static_cast<size_t>(j)] +
                   combinations[static_cast<size_t>(j - 1)]) %
                  mod_value;
          }
      }
      long long answer = combinations[static_cast<size_t>(n)] * power_mod(a, n) % mod_value;
      answer = answer * power_mod(b, m) % mod_value;
      cout << answer << '\n';
      return 0;
  }
