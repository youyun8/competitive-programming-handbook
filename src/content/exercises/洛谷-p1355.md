cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;

  struct Point {
      long long x;
      long long y;
  };

  static bool same_point(const Point& a, const Point& b) {
      return a.x == b.x && a.y == b.y;
  }

  static long long cross(const Point& a, const Point& b, const Point& p) {
      return (b.x - a.x) * (p.y - a.y) - (b.y - a.y) * (p.x - a.x);
  }

  static bool on_segment(const Point& a, const Point& b, const Point& p) {
      return cross(a, b, p) == 0 &&
             min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
             min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
  }

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      array<Point, 3> triangle{};
      Point point{};
      for (Point& vertex : triangle) { cin >> vertex.x >> vertex.y; }
      cin >> point.x >> point.y;
      for (const Point& vertex : triangle) {
          if (same_point(vertex, point)) {
              cout << 4 << '\n';
              return 0;
          }
      }
      for (int i = 0; i < 3; ++i) {
          if (on_segment(triangle[static_cast<size_t>(i)],
                         triangle[static_cast<size_t>((i + 1) % 3)], point)) {
              cout << 3 << '\n';
              return 0;
          }
      }
      bool has_positive = false;
      bool has_negative = false;
      for (int i = 0; i < 3; ++i) {
          const long long value =
              cross(triangle[static_cast<size_t>(i)],
                    triangle[static_cast<size_t>((i + 1) % 3)], point);
          has_positive = has_positive || value > 0;
          has_negative = has_negative || value < 0;
      }
      cout << (has_positive && has_negative ? 2 : 1) << '\n';
  }
