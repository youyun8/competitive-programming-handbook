cpp_solution: |
  #include <algorithm>
  #include <iostream>
  #include <vector>
  using namespace std;
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int n;
      cin >> n;
      vector<long long> happy(static_cast<size_t>(n + 1));
      for (int i = 1; i <= n; ++i) cin >> happy[i];
      vector<vector<int>> children(static_cast<size_t>(n + 1));
      vector<char> has_parent(static_cast<size_t>(n + 1));
      int employee, boss;
      while (cin >> employee >> boss && (employee != 0 || boss != 0)) {
          children[boss].push_back(employee);
          has_parent[employee] = 1;
      }
      int root = 1;
      while (has_parent[root] != 0) ++root;
      vector<long long> skip(static_cast<size_t>(n + 1));
      vector<long long> take = happy;
      auto dfs = [&](auto&& self, int u) -> void {
          for (int v : children[u]) {
              self(self, v);
              take[u] += skip[v];
              skip[u] += max(skip[v], take[v]);
          }
      };
      dfs(dfs, root);
      cout << max(skip[root], take[root]) << '\n';
  }
