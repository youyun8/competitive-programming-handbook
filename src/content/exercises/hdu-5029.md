statement: 多組測資。每次救濟在樹上 x-y 路徑的每個村莊發放一次種類 z 的糧食。全部 m 次發放後，對每個村莊輸出收到次數最多的糧食種類；並列時選種類編號最小者，從未收到則輸出 0。
constraints: ['1 <= n <= 100000', '0 <= m <= 100000', '1 <= z <= 100000', '測資不超過 25 組，以 0 0 結束']
input_format: 每組先給 n、m；接著 n-1 條樹邊與 m 行 x、y、z。
output_format: 每組輸出 n 行，第 i 行是村莊 i 的答案。
samples:
  - input: |
      3 2
      1 2
      2 3
      1 2 5
      2 3 7
      0 0
    output: |
      5
      5
      7
    explanation: 村莊 2 同時收到 5、7 各一次，依規則選較小的 5；另外兩村只收到各自一種。
core_knowledge: [重鏈剖分將路徑拆為區間, 區間差分事件, 種類計數最大值線段樹]
judgment: 求的是發放「次數」最多的種類，不是種類編號最大；並列取最小編號。
hints:
  - 樹剖後，每次路徑可拆成 O(log n) 個 DFS 序閉區間。
  - 對每個 `[l,r]`，在位置 l 加入種類 z、位置 r+1 移除種類 z；掃描 DFS 序即可維護當前點收到的所有種類次數。
  - 以種類為線段樹下標，葉節點存目前次數；父節點保留次數較大者，相同時保留左側較小種類。
solution_outline: 重鏈剖分所有路徑，建立區間端點事件。依 DFS 序掃描，將當前位置事件轉成種類計數單點增減；線段樹根即目前最高次數與最小種類，映回原節點。
proof_or_invariant: 差分掃描至位置 p 時，種類 z 的計數等於所有涵蓋 p 的 z 區間數；樹剖區間聯集恰為每條原路徑。種類線段樹依最大次數、最小編號的順序合併，故根節點正是該村答案。
common_errors: [右端移除事件放在 r 而非 r+1, 並列時選到較大種類, 掃 DFS 序後忘記映回原節點]
complexity: { time: 'O((n+m log n)log Z)，Z<=100000', space: 'O(n+m log n+Z)' }

cpp_solution: |
  #include <algorithm>
  #include <iostream>
  #include <utility>
  #include <vector>
  using namespace std;
  struct ModeTree {
      int size = 1;
      vector<int> count, color;
      explicit ModeTree(int n) {
          while (size < n) size *= 2;
          count.assign(static_cast<size_t>(2 * size), 0);
          color.assign(static_cast<size_t>(2 * size), 0);
          for (int i = 0; i < size; ++i) color[static_cast<size_t>(size + i)] = i + 1;
          for (int node = size - 1; node > 0; --node) color[static_cast<size_t>(node)] = color[static_cast<size_t>(node * 2)];
      }
      void update(int position, int delta) {
          int node = size + position - 1;
          count[static_cast<size_t>(node)] += delta;
          for (node /= 2; node > 0; node /= 2) {
              const int left = node * 2, right = left + 1;
              if (count[static_cast<size_t>(left)] >= count[static_cast<size_t>(right)]) {
                  count[static_cast<size_t>(node)] = count[static_cast<size_t>(left)];
                  color[static_cast<size_t>(node)] = color[static_cast<size_t>(left)];
              } else {
                  count[static_cast<size_t>(node)] = count[static_cast<size_t>(right)];
                  color[static_cast<size_t>(node)] = color[static_cast<size_t>(right)];
              }
          }
      }
      int answer() const { return count[1] == 0 ? 0 : color[1]; }
  };
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int n, m;
      while (cin >> n >> m && (n != 0 || m != 0)) {
          vector<vector<int>> graph(static_cast<size_t>(n) + 1U);
          for (int i = 1; i < n; ++i) { int u, v; cin >> u >> v; graph[static_cast<size_t>(u)].push_back(v); graph[static_cast<size_t>(v)].push_back(u); }
          vector<int> parent(static_cast<size_t>(n) + 1U), depth(static_cast<size_t>(n) + 1U), order{1}; depth[1] = 1;
          for (size_t i = 0; i < order.size(); ++i) for (int next : graph[static_cast<size_t>(order[i])]) if (next != parent[static_cast<size_t>(order[i])]) { parent[static_cast<size_t>(next)] = order[i]; depth[static_cast<size_t>(next)] = depth[static_cast<size_t>(order[i])] + 1; order.push_back(next); }
          vector<int> subtree_size(static_cast<size_t>(n) + 1U, 1), heavy(static_cast<size_t>(n) + 1U);
          for (size_t i = order.size(); i-- > 0;) for (int next : graph[static_cast<size_t>(order[i])]) if (parent[static_cast<size_t>(next)] == order[i]) { subtree_size[static_cast<size_t>(order[i])] += subtree_size[static_cast<size_t>(next)]; if (subtree_size[static_cast<size_t>(next)] > subtree_size[static_cast<size_t>(heavy[static_cast<size_t>(order[i])])]) heavy[static_cast<size_t>(order[i])] = next; }
          vector<int> head(static_cast<size_t>(n) + 1U), position(static_cast<size_t>(n) + 1U), node_at(static_cast<size_t>(n) + 1U);
          vector<pair<int, int>> stack{{1, 1}}; int timer = 0;
          while (!stack.empty()) { auto [start, chain_head] = stack.back(); stack.pop_back(); for (int node = start; node != 0; node = heavy[static_cast<size_t>(node)]) { head[static_cast<size_t>(node)] = chain_head; position[static_cast<size_t>(node)] = ++timer; node_at[static_cast<size_t>(timer)] = node; for (int next : graph[static_cast<size_t>(node)]) if (parent[static_cast<size_t>(next)] == node && next != heavy[static_cast<size_t>(node)]) stack.push_back({next, next}); } }
          vector<vector<int>> events(static_cast<size_t>(n) + 2U);
          int maximum_color = 1;
          auto add_interval = [&](int left, int right, int color) { events[static_cast<size_t>(left)].push_back(color); events[static_cast<size_t>(right + 1)].push_back(-color); };
          while (m--) {
              int x, y, color; cin >> x >> y >> color; maximum_color = max(maximum_color, color);
              while (head[static_cast<size_t>(x)] != head[static_cast<size_t>(y)]) {
                  if (depth[static_cast<size_t>(head[static_cast<size_t>(x)])] < depth[static_cast<size_t>(head[static_cast<size_t>(y)])]) swap(x, y);
                  add_interval(position[static_cast<size_t>(head[static_cast<size_t>(x)])], position[static_cast<size_t>(x)], color);
                  x = parent[static_cast<size_t>(head[static_cast<size_t>(x)])];
              }
              if (position[static_cast<size_t>(x)] > position[static_cast<size_t>(y)]) swap(x, y);
              add_interval(position[static_cast<size_t>(x)], position[static_cast<size_t>(y)], color);
          }
          ModeTree mode(maximum_color);
          vector<int> answer(static_cast<size_t>(n) + 1U);
          for (int pos = 1; pos <= n; ++pos) {
              for (int event : events[static_cast<size_t>(pos)])
                  mode.update(abs(event), event > 0 ? 1 : -1);
              answer[static_cast<size_t>(node_at[static_cast<size_t>(pos)])] = mode.answer();
          }
          for (int node = 1; node <= n; ++node) cout << answer[static_cast<size_t>(node)] << '\n';
      }
      return 0;
  }
