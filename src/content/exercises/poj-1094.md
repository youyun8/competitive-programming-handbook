solution_outline: >-
  以鄰接矩陣去除重複邊。每次加邊後複製入度執行 Kahn：掃描未取且入度零的候選數；
  零個回報矛盾，多於一個標記不唯一，恰一個則接到結果。保存第一次非不確定狀態。
proof_or_invariant: >-
  每一步零入度點恰是所有前置元素已放入的合法下一項。候選至少兩個時，分別先選任一個
  都可延伸為拓撲序，故不唯一；每步唯一則所有合法序列逐位被迫相同。若剩餘子圖無零入度
  點，沿前驅必重訪而有環。逐條檢查所以首次狀態的編號正確。

cpp_solution: |
  #include <iostream>
  #include <string>
  #include <vector>
  using namespace std;
  struct Result{int state=0;string order;};
  static Result check(const vector<vector<bool>>& edge,const vector<int>& original){
      int n=static_cast<int>(edge.size());vector<int> degree=original;vector<bool> used(edge.size(),false);string order;bool unique=true;
      for(int position=0;position<n;++position){int candidate=-1,count=0;for(int v=0;v<n;++v)if(!used[static_cast<size_t>(v)]&&degree[static_cast<size_t>(v)]==0){candidate=v;++count;}if(count==0)return {-1,""};if(count>1)unique=false;used[static_cast<size_t>(candidate)]=true;order.push_back(static_cast<char>('A'+candidate));for(int v=0;v<n;++v)if(edge[static_cast<size_t>(candidate)][static_cast<size_t>(v)])--degree[static_cast<size_t>(v)];}
      return {unique?1:0,order};
  }
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);int n=0,m=0;
      while(cin>>n>>m&&(n!=0||m!=0)){
          vector<vector<bool>> edge(static_cast<size_t>(n),vector<bool>(static_cast<size_t>(n),false));vector<int> indegree(static_cast<size_t>(n),0);int answer_state=0,answer_at=0;string answer_order;
          for(int relation_index=1;relation_index<=m;++relation_index){string relation;cin>>relation;if(answer_state!=0)continue;int from=relation[0]-'A',to=relation[2]-'A';if(!edge[static_cast<size_t>(from)][static_cast<size_t>(to)]){edge[static_cast<size_t>(from)][static_cast<size_t>(to)]=true;++indegree[static_cast<size_t>(to)];}Result result=check(edge,indegree);if(result.state!=0){answer_state=result.state;answer_at=relation_index;answer_order=result.order;}}
          if(answer_state==-1)cout<<"Inconsistency found after "<<answer_at<<" relations.\n";else if(answer_state==1)cout<<"Sorted sequence determined after "<<answer_at<<" relations: "<<answer_order<<".\n";else cout<<"Sorted sequence cannot be determined.\n";
      }
  }
