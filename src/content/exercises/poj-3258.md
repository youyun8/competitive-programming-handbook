cpp_solution: |
  #include <algorithm>
  #include <iostream>
  #include <vector>
  using namespace std;

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int river_length = 0;
      int rock_count = 0;
      int removal_limit = 0;
      cin >> river_length >> rock_count >> removal_limit;
      vector<int> positions(static_cast<size_t>(rock_count) + 2U);
      positions[0] = 0;
      positions.back() = river_length;
      for (int i = 1; i <= rock_count; ++i) {
          cin >> positions[static_cast<size_t>(i)];
      }
      sort(positions.begin(), positions.end());

      const auto is_feasible = [&](int minimum_jump) {
          int removed = 0;
          int last_position = positions[0];
          for (size_t i = 1; i < positions.size(); ++i) {
              if (positions[i] - last_position < minimum_jump) {
                  ++removed;
              } else {
                  last_position = positions[i];
              }
          }
          return removed <= removal_limit;
      };

      int low = 0;
      int high = river_length + 1;
      while (low + 1 < high) {
          const int middle = low + (high - low) / 2;
          if (is_feasible(middle)) {
              low = middle;
          } else {
              high = middle;
          }
      }
      cout << low << '\n';
      return 0;
  }
