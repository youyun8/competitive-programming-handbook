proof_or_invariant: 對任意 mask，存在 AND 涵蓋 mask 的生成樹，若且唯若由所有權重涵蓋 mask 的邊形成的子圖連通。演算法按二進位字典序由高位到低位固定可行位；高位的價值大於所有較低位之和，因此每次接受可行高位都不會排除更佳答案。歸納至最低位即得最大值。

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  struct Edge { int u, v, w; };
  class DisjointSet {
  public:
      explicit DisjointSet(int n) : parent_(static_cast<size_t>(n + 1)), size_(static_cast<size_t>(n + 1), 1) {
          iota(parent_.begin(), parent_.end(), 0);
      }
      int find_root(int x) {
          if (parent_[static_cast<size_t>(x)] != x) parent_[static_cast<size_t>(x)] = find_root(parent_[static_cast<size_t>(x)]);
          return parent_[static_cast<size_t>(x)];
      }
      bool unite(int x, int y) {
          x = find_root(x); y = find_root(y);
          if (x == y) return false;
          if (size_[static_cast<size_t>(x)] < size_[static_cast<size_t>(y)]) swap(x, y);
          parent_[static_cast<size_t>(y)] = x;
          size_[static_cast<size_t>(x)] += size_[static_cast<size_t>(y)];
          return true;
      }
  private:
      vector<int> parent_, size_;
  };
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int tests; cin >> tests;
      while (tests--) {
          int n, m; cin >> n >> m;
          vector<Edge> edges(static_cast<size_t>(m));
          for (auto& edge : edges) cin >> edge.u >> edge.v >> edge.w;
          int answer = 0;
          for (int bit = 30; bit >= 0; --bit) {
              const int candidate = answer | (1 << bit);
              DisjointSet dsu(n);
              int merged = 0;
              for (const auto& edge : edges)
                  if ((edge.w & candidate) == candidate && dsu.unite(edge.u, edge.v)) ++merged;
              if (merged == n - 1) answer = candidate;
          }
          cout << answer << '\n';
      }
  }
