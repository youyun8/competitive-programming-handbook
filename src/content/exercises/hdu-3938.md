proof_or_invariant: 加完所有 w<=L 的邊後，兩點同分量當且僅當存在一條所有邊皆不超過 L 的路，亦即其 minimax 成本不超過 L。合併兩個分量時，唯一新增的合法無序點對恰為跨兩分量的笛卡兒積，共 a*b 對；各對只在首次連通時被計一次，因此累計值正確。

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  struct Edge { int u, v, w; };
  struct Query { int limit, id; };
  class DisjointSet {
  public:
      explicit DisjointSet(int n) : parent_(static_cast<size_t>(n + 1)), size_(static_cast<size_t>(n + 1), 1) { iota(parent_.begin(), parent_.end(), 0); }
      int find_root(int x) { return parent_[static_cast<size_t>(x)] == x ? x : parent_[static_cast<size_t>(x)] = find_root(parent_[static_cast<size_t>(x)]); }
      long long unite(int x, int y) {
          x = find_root(x); y = find_root(y);
          if (x == y) return 0;
          if (size_[static_cast<size_t>(x)] < size_[static_cast<size_t>(y)]) swap(x, y);
          const long long added = 1LL * size_[static_cast<size_t>(x)] * size_[static_cast<size_t>(y)];
          parent_[static_cast<size_t>(y)] = x; size_[static_cast<size_t>(x)] += size_[static_cast<size_t>(y)]; return added;
      }
  private: vector<int> parent_, size_;
  };
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int n, m, q;
      while (cin >> n >> m >> q) {
          vector<Edge> edges(static_cast<size_t>(m));
          for (auto& edge : edges) cin >> edge.u >> edge.v >> edge.w;
          vector<Query> queries(static_cast<size_t>(q));
          for (int i = 0; i < q; ++i) { cin >> queries[static_cast<size_t>(i)].limit; queries[static_cast<size_t>(i)].id = i; }
          sort(edges.begin(), edges.end(), [](const Edge& a, const Edge& b) { return a.w < b.w; });
          sort(queries.begin(), queries.end(), [](const Query& a, const Query& b) { return a.limit < b.limit; });
          DisjointSet dsu(n);
          vector<long long> answer(static_cast<size_t>(q));
          long long pairs = 0; size_t edge_index = 0;
          for (const auto& query : queries) {
              while (edge_index < edges.size() && edges[edge_index].w <= query.limit) {
                  pairs += dsu.unite(edges[edge_index].u, edges[edge_index].v); ++edge_index;
              }
              answer[static_cast<size_t>(query.id)] = pairs;
          }
          for (const long long value : answer) cout << value << '\n';
      }
  }
