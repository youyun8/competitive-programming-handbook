statement: 依序排版 N 個不可拆分單字，頁寬 W，同一連續文字區內相鄰單字需一格空白。另有寬 pw、左邊距 dw 的圖片；每次詢問指定圖片起始行 x 與高度 h，求文字和圖片最少共占幾行。
constraints: ['T ≤ 10', '1 ≤ N,W,Q ≤ 100000', '1 ≤ pw,a_i ≤ W', '0 ≤ dw ≤ W-pw']
input_format: 每組輸入 N、W、pw、dw；下一行 N 個字長；再輸入 Q 與 Q 行 x、h。
output_format: 每個詢問輸出最少占用行數。
samples:
  - input: |
      1
      2 7 4 3
      1 3
      1
      1 1
    output: |
      2
    explanation: 圖片行左側寬 3，只能放第一個字；第二個字須在下一個完整寬度行，合計兩行。
core_knowledge: ['預處理一行後的首個未排單字', '完整行與圖片行兩種轉移倍增']
judgment: 對固定可用寬度與起始單字，貪心塞入最長前綴必然最少行；圖片行依序使用左、右兩個文字區。
hints:
  - '用前綴和求從單字 i 開始，在寬 width 中最多放到哪裡。'
  - '建立 normal[i]；圖片行轉移為先套寬 dw，再套寬 W-pw-dw。'
  - '對兩種轉移建倍增表；詢問依序跳 x-1 個完整行、h 個圖片行，再加剩餘完整行數。'
solution_outline: 以字長加一的前綴和及 upper_bound 建三種寬度轉移。組合左右寬為 picture transition，對 normal、picture 各建倍增表，另倒推 normal_tail。每詢問 O(log N) 套用三階段。
proof_or_invariant: 每行放可容納的最長單字前綴不會使後續比任何較短放法更晚，故單行轉移最優。圖片左右區按閱讀順序連續，組合轉移精確。倍增僅合成確定性轉移，三階段與頁面區域逐行一致，因此行數最少。
common_errors: ['允許單字跨越圖片本體', '左右文字區之間錯加空白', '忘記圖片本身占行', 'h 次轉移逐次模擬造成超時']
complexity: { time: 'O((N+Q) log N)', space: 'O(N log N)' }

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int tests;cin>>tests;
      while(tests--){
          int n,w,picture_width,left_width;cin>>n>>w>>picture_width>>left_width;
          vector<long long> prefix(static_cast<size_t>(n)+1U);
          for(int i=0;i<n;++i){int length;cin>>length;prefix[static_cast<size_t>(i+1)]=prefix[static_cast<size_t>(i)]+length+1;}
          const auto build_next=[&](int width){vector<int> next(static_cast<size_t>(n)+1U,n);for(int i=0;i<n;++i){const long long limit=prefix[static_cast<size_t>(i)]+width+1;next[static_cast<size_t>(i)]=static_cast<int>(upper_bound(prefix.begin()+i+1,prefix.end(),limit)-prefix.begin())-1;}return next;};
          vector<int> normal=build_next(w),left=build_next(left_width),right=build_next(w-picture_width-left_width),picture(static_cast<size_t>(n)+1U,n);
          for(int i=0;i<n;++i)picture[static_cast<size_t>(i)]=right[static_cast<size_t>(left[static_cast<size_t>(i)])];
          constexpr int levels=20;
          vector<vector<int>> normal_jump(levels,vector<int>(static_cast<size_t>(n)+1U,n)),picture_jump=normal_jump;
          normal_jump[0]=normal;picture_jump[0]=picture;
          for(int k=1;k<levels;++k)for(int i=0;i<=n;++i){normal_jump[static_cast<size_t>(k)][static_cast<size_t>(i)]=normal_jump[static_cast<size_t>(k-1)][static_cast<size_t>(normal_jump[static_cast<size_t>(k-1)][static_cast<size_t>(i)])];picture_jump[static_cast<size_t>(k)][static_cast<size_t>(i)]=picture_jump[static_cast<size_t>(k-1)][static_cast<size_t>(picture_jump[static_cast<size_t>(k-1)][static_cast<size_t>(i)])];}
          vector<int> tail(static_cast<size_t>(n)+1U);for(int i=n-1;i>=0;--i)tail[static_cast<size_t>(i)]=1+tail[static_cast<size_t>(normal[static_cast<size_t>(i)])];const int total_lines=tail[0];
          const auto advance=[&](int position,int count,const vector<vector<int>>&jump){for(int k=0;k<levels;++k)if((count&(1<<k))!=0)position=jump[static_cast<size_t>(k)][static_cast<size_t>(position)];return position;};
          int queries;cin>>queries;while(queries--){int x,h;cin>>x>>h;const int before=x-1;if(before>=total_lines){cout<<total_lines+h<<'\n';continue;}int position=advance(0,before,normal_jump);position=advance(position,h,picture_jump);if(position==n)cout<<before+h<<'\n';else cout<<before+h+tail[static_cast<size_t>(position)]<<'\n';}
      }
  }
