hints:
  - 先列出不超過 1000 的 Fibonacci 數。
  - sg[x] 是所有 sg[x-f]（f<=x）的 mex。
  - 三堆組合局面的 SG 值是 sg[m] xor sg[n] xor sg[p]。
solution_outline: 自底向上計算 sg[0..1000]，每個 x 標記所有合法後繼值並取 mex；查詢 XOR 是否非零。
proof_or_invariant: 所有操作都使單堆大小下降，故計算 sg[x] 時後繼已知，mex 定義正確。三堆操作集合是三個互不干擾子遊戲的直和，Sprague-Grundy 定理給其值為 XOR；零值恰為必敗態。
common_errors: [使用 1, 1, 2 開頭的 Fibonacci 而重複無妨但漏掉 2, 把三個 SG 相加, 終止行仍輸出答案]

cpp_solution: |
  #include <array>
  #include <iostream>
  #include <vector>
  using namespace std;
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      vector<int> fibonacci{1, 2};
      while (fibonacci.back() <= 1000)
          fibonacci.push_back(fibonacci[fibonacci.size() - 1U] +
                              fibonacci[fibonacci.size() - 2U]);
      array<int, 1001> grundy{};
      for (int stones = 1; stones <= 1000; ++stones) {
          array<bool, 32> seen{};
          for (int take : fibonacci) {
              if (take > stones) break;
              seen[static_cast<size_t>(grundy[static_cast<size_t>(stones - take)])] = true;
          }
          int value = 0;
          while (seen[static_cast<size_t>(value)]) ++value;
          grundy[static_cast<size_t>(stones)] = value;
      }
      int first, second, third;
      while (cin >> first >> second >> third &&
             (first != 0 || second != 0 || third != 0)) {
          const int xor_sum = grundy[static_cast<size_t>(first)] ^
                              grundy[static_cast<size_t>(second)] ^
                              grundy[static_cast<size_t>(third)];
          cout << (xor_sum != 0 ? "Fibo\n" : "Nacci\n");
      }
      return 0;
  }
