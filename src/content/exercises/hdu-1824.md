  #include <bits/stdc++.h>
  using namespace std;
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int teams,m;while(cin>>teams>>m){vector<int> person_team(static_cast<size_t>(3*teams)),person_value(static_cast<size_t>(3*teams));for(int i=0;i<teams;++i){int captain,a,b;cin>>captain>>a>>b;person_team[static_cast<size_t>(captain)]=i;person_value[static_cast<size_t>(captain)]=1;person_team[static_cast<size_t>(a)]=person_team[static_cast<size_t>(b)]=i;person_value[static_cast<size_t>(a)]=person_value[static_cast<size_t>(b)]=0;}
          vector<vector<int>> graph(static_cast<size_t>(2*teams));auto literal=[&](int person){return 2*person_team[static_cast<size_t>(person)]+person_value[static_cast<size_t>(person)];};
          for(int i=0;i<m;++i){int a,b;cin>>a>>b;int x=literal(a),y=literal(b);graph[static_cast<size_t>(x)].push_back(y^1);graph[static_cast<size_t>(y)].push_back(x^1);}
          vector<int> dfn(static_cast<size_t>(2*teams)),low(static_cast<size_t>(2*teams)),component(static_cast<size_t>(2*teams),-1),stack_nodes;vector<char> on_stack(static_cast<size_t>(2*teams));int timer=0,count=0;
          auto dfs=[&](auto&& self,int u)->void{dfn[static_cast<size_t>(u)]=low[static_cast<size_t>(u)]=++timer;stack_nodes.push_back(u);on_stack[static_cast<size_t>(u)]=1;for(int v:graph[static_cast<size_t>(u)])if(!dfn[static_cast<size_t>(v)]){self(self,v);low[static_cast<size_t>(u)]=min(low[static_cast<size_t>(u)],low[static_cast<size_t>(v)]);}else if(on_stack[static_cast<size_t>(v)])low[static_cast<size_t>(u)]=min(low[static_cast<size_t>(u)],dfn[static_cast<size_t>(v)]);if(low[static_cast<size_t>(u)]==dfn[static_cast<size_t>(u)]){while(true){int v=stack_nodes.back();stack_nodes.pop_back();on_stack[static_cast<size_t>(v)]=0;component[static_cast<size_t>(v)]=count;if(v==u)break;}++count;}};
          for(int i=0;i<2*teams;++i){if(!dfn[static_cast<size_t>(i)])dfs(dfs,i);}
          bool possible=true;for(int i=0;i<teams;++i){if(component[static_cast<size_t>(2*i)]==component[static_cast<size_t>(2*i+1)])possible=false;}
          cout<<(possible?"yes":"no")<<'\n';
      }
  }
