cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int n;while(cin>>n){
          vector<vector<double>> safe(static_cast<size_t>(n),vector<double>(static_cast<size_t>(n)));
          for(auto& row:safe)for(double& value:row)cin>>value;
          for(int k=0;k<n;++k)for(int i=0;i<n;++i)for(int j=0;j<n;++j)
              safe[static_cast<size_t>(i)][static_cast<size_t>(j)]=max(
                  safe[static_cast<size_t>(i)][static_cast<size_t>(j)],
                  safe[static_cast<size_t>(i)][static_cast<size_t>(k)]*safe[static_cast<size_t>(k)][static_cast<size_t>(j)]);
          int q;cin>>q;while(q-->0){int a,b;cin>>a>>b;double answer=safe[static_cast<size_t>(a-1)][static_cast<size_t>(b-1)];
              if(answer==0)cout<<"What a pity!\n";else cout<<fixed<<setprecision(3)<<answer<<'\n';}
      }
  }
