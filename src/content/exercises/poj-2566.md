solution_outline: 建立 prefix[0..n]，每項保存總和與位置，依總和排序。對每個 t 用兩個不同位置的指標掃描：比較 prefix[high].sum-prefix[low].sum 與 t，記錄距離更小者；差偏小移動 high，否則移動 low，兩指標相等時先推進 high。答案端點由兩個原前綴位置的較小者加一與較大者取得。
proof_or_invariant: '任一非空區間 [l,r] 對應兩個不同前綴位置 l-1、r，其絕對和正是排序後這兩值的大者減小者，反之亦然。對排序陣列的 low、high，若差<t，固定 low 時更小的 high 不可能更接近，故可增加 high；若差≥t，固定 high 時更小的 low 只會令差更大，故可增加 low。被略過的配對都不可能優於移動方向上的下一候選，因此掃描會考慮到一個全域最近差。'

cpp_solution: |
  #include <algorithm>
  #include <cstdlib>
  #include <iostream>
  #include <limits>
  #include <utility>
  #include <vector>
  using namespace std;

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int n = 0;
      int query_count = 0;
      while (cin >> n >> query_count && (n != 0 || query_count != 0)) {
          vector<pair<long long, int>> prefix(static_cast<size_t>(n) + 1U);
          prefix[0] = {0, 0};
          for (int i = 1; i <= n; ++i) {
              long long value = 0;
              cin >> value;
              prefix[static_cast<size_t>(i)] = {
                  prefix[static_cast<size_t>(i - 1)].first + value,
                  i
              };
          }
          sort(prefix.begin(), prefix.end(), [](const auto& first, const auto& second) {
              if (first.first != second.first) {
                  return first.first < second.first;
              }
              return first.second > second.second;
          });

          while (query_count-- > 0) {
              long long target = 0;
              cin >> target;
              size_t low = 0;
              size_t high = 1;
              long long best_distance = numeric_limits<long long>::max();
              long long best_sum = 0;
              int best_left = 1;
              int best_right = 1;
              while (high < prefix.size()) {
                  if (low == high) {
                      ++high;
                      continue;
                  }
                  const long long current_sum = prefix[high].first - prefix[low].first;
                  const long long distance = llabs(current_sum - target);
                  if (distance < best_distance) {
                      best_distance = distance;
                      best_sum = current_sum;
                      const int first_position = prefix[low].second;
                      const int second_position = prefix[high].second;
                      best_left = min(first_position, second_position) + 1;
                      best_right = max(first_position, second_position);
                  }
                  if (current_sum < target) {
                      ++high;
                  } else {
                      ++low;
                  }
              }
              cout << best_sum << ' ' << best_left << ' ' << best_right << '\n';
          }
      }
      return 0;
  }
