hints:
  - 第 i 個單詞終點標記位元 1<<i，並沿 fail 以 OR 傳遞。
  - dp[node][mask] 表示目前 AC 狀態及已見單詞集合的方案數。
  - 每加一字元轉移到 v，令新遮罩為 mask|output[v]；末輪統計 popcount>=k。
solution_outline: 建 AC 並傳遞輸出遮罩；從 dp[root][0]=1 做 n 輪 26 字元轉移，最後加總至少 k 位為一的狀態。
proof_or_invariant: 遮罩精確記錄前綴中曾結束的所有模式，AC 狀態精確記錄其最長可用後綴；每個長 n 字串有唯一轉移路徑，因此末態條件與題意等價且無重複。
common_errors: [輸出遮罩未繼承 fail, 用出現次數取代不同種類, 忘記每次加法取模]

  #include <algorithm>
  #include <array>
  #include <iostream>
  #include <queue>
  #include <string>
  #include <vector>
  using namespace std;
  constexpr int mod=20090717;
  struct Node{array<int,26> next{};int fail=0;int output=0;};
  static int bit_count(unsigned value){int result=0;while(value!=0U){result+=static_cast<int>(value&1U);value>>=1U;}return result;}
  int main(){ios::sync_with_stdio(false);cin.tie(nullptr);int n=0,m=0,k=0;while(cin>>n>>m>>k&&(n!=0||m!=0||k!=0)){vector<Node>a(1);for(int i=0;i<m;++i){string s;cin>>s;int u=0;for(char ch:s){size_t c=static_cast<size_t>(ch-'a');if(a[static_cast<size_t>(u)].next[c]==0){a[static_cast<size_t>(u)].next[c]=static_cast<int>(a.size());a.push_back({});}u=a[static_cast<size_t>(u)].next[c];}a[static_cast<size_t>(u)].output|=1<<i;}queue<int>q;for(size_t c=0;c<26;++c)if(a[0].next[c]!=0)q.push(a[0].next[c]);while(!q.empty()){int u=q.front();q.pop();a[static_cast<size_t>(u)].output|=a[static_cast<size_t>(a[static_cast<size_t>(u)].fail)].output;for(size_t c=0;c<26;++c){int&v=a[static_cast<size_t>(u)].next[c];if(v!=0){a[static_cast<size_t>(v)].fail=a[static_cast<size_t>(a[static_cast<size_t>(u)].fail)].next[c];q.push(v);}else v=a[static_cast<size_t>(a[static_cast<size_t>(u)].fail)].next[c];}}size_t masks=static_cast<size_t>(1U<<m);vector<vector<int>>dp(a.size(),vector<int>(masks)),next=dp;dp[0][0]=1;for(int length=0;length<n;++length){for(auto&row:next)fill(row.begin(),row.end(),0);for(size_t u=0;u<a.size();++u)for(size_t mask=0;mask<masks;++mask)if(dp[u][mask]!=0)for(size_t c=0;c<26;++c){int v=a[u].next[c];size_t nm=mask|static_cast<size_t>(a[static_cast<size_t>(v)].output);next[static_cast<size_t>(v)][nm]=(next[static_cast<size_t>(v)][nm]+dp[u][mask])%mod;}dp.swap(next);}int answer=0;for(const auto&row:dp)for(size_t mask=0;mask<masks;++mask)if(bit_count(static_cast<unsigned>(mask))>=k)answer=(answer+row[mask])%mod;cout<<answer<<'\n';}}
