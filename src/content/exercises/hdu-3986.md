statement: 多組無向正權圖，敵人可破壞一條道路；求從 1 到 n 在最壞破壞下的最短距離，若能使兩點不連通輸出 -1。
constraints: [邊權為正, 可能有重邊, 多組測試]
input_format: 第一行 T；每組輸入 n、m，再輸入 m 條 u、v、w。
output_format: 每組輸出刪除一邊後最短距離的最大值，若某次不可達輸出 -1。
samples:
  - input: |-
      1
      3 3
      1 2 1
      2 3 1
      1 3 5
    output: '5'
    explanation: 破壞最短路上的任一條權 1 道路後，只能走權 5 的直接道路。
hints:
  - 先跑一次 Dijkstra 並保存一條 1 到 n 最短路的邊編號。
  - 只枚舉刪除這條路上的邊；刪除其他邊不會增加距離。
  - 每次以邊編號略過單一無向邊，重邊不能只用端點識別。
solution_outline: 初次 Dijkstra 回溯出一條最短路；依次禁用路上每條邊重跑 Dijkstra，若不可達立即 -1，否則取距離最大值。
proof_or_invariant: 若被刪邊不在固定最短路 P，P 仍證明新距離不大於原距離，而刪邊不會縮短距離，故距離不變。會使答案增加的邊必在 P，枚舉 P 即完整。
complexity: { time: 'O(n(m+n) log n)', space: 'O(n+m)' }
common_errors: [枚舉所有 m 條邊造成不必要逾時, 重邊以端點刪掉多條, 不可達時仍輸出無限值]

  struct Edge{int to,weight,id;};
  int main(){

      int tests;if(!(cin>>tests))return 0;while(tests-->0){int n,m;cin>>n>>m;vector<vector<Edge>> graph(static_cast<size_t>(n+1));for(int id=0;id<m;++id){int u,v,w;cin>>u>>v>>w;graph[static_cast<size_t>(u)].push_back({v,w,id});graph[static_cast<size_t>(v)].push_back({u,w,id});}
          const long long inf=LLONG_MAX/4;auto dijkstra=[&](int banned,vector<int>* parent){vector<long long> dist(static_cast<size_t>(n+1),inf);priority_queue<pair<long long,int>,vector<pair<long long,int>>,greater<>> queue;dist[1]=0;queue.push({0,1});if(parent)parent->assign(static_cast<size_t>(n+1),-1);while(!queue.empty()){auto[d,u]=queue.top();queue.pop();if(d!=dist[static_cast<size_t>(u)])continue;for(const Edge& e:graph[static_cast<size_t>(u)])if(e.id!=banned&&d+e.weight<dist[static_cast<size_t>(e.to)]){dist[static_cast<size_t>(e.to)]=d+e.weight;if(parent)(*parent)[static_cast<size_t>(e.to)]=e.id;queue.push({d+e.weight,e.to});}}return dist[static_cast<size_t>(n)];};
          vector<int> parent_edge;long long original=dijkstra(-1,&parent_edge);if(original>=inf){cout<<"-1\n";continue;}vector<int> path;int current=n;while(current!=1){int id=parent_edge[static_cast<size_t>(current)];path.push_back(id);for(const Edge& e:graph[static_cast<size_t>(current)])if(e.id==id){current=e.to;break;}}
          long long answer=original;bool disconnected=false;for(int id:path){long long value=dijkstra(id,nullptr);if(value>=inf){disconnected=true;break;}answer=max(answer,value);}cout<<(disconnected?-1:answer)<<'\n';
      }
  }
