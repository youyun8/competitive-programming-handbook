cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;
  int main() {
      ios::sync_with_stdio(false); cin.tie(nullptr);
      int tests; cin >> tests;
      vector<int> query(static_cast<size_t>(tests));
      int maximum = 0;
      for (int& value : query) { cin >> value; maximum = max(maximum, value); }
      const int limit = 3 * maximum + 7;
      vector<char> prime(static_cast<size_t>(limit) + 1, true);
      if (limit >= 0) prime[0] = false;
      if (limit >= 1) prime[1] = false;
      for (int p = 2; p <= limit / p; ++p)
          if (prime[static_cast<size_t>(p)])
              for (int multiple = p * p; multiple <= limit; multiple += p)
                  prime[static_cast<size_t>(multiple)] = false;
      vector<int> prefix(static_cast<size_t>(maximum) + 1, 0);
      for (int k = 1; k <= maximum; ++k)
          prefix[static_cast<size_t>(k)] = prefix[static_cast<size_t>(k - 1)] +
              (prime[static_cast<size_t>(3 * k + 7)] ? 1 : 0);
      for (int n : query) cout << prefix[static_cast<size_t>(n)] << '\n';
      return 0;
  }
