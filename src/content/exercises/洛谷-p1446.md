hints:
  - 把每個洗牌分解成循環，循環長度就是一次不可拆分的染色份量。
  - 用 dp[r][b][g] 逐循環選紅、藍、綠，算出恰用完指定數量的固定染色。
  - 別忘記恆等置換；固定點總和乘 (m+1) 在模 P 下的逆元。
solution_outline: 對輸入 m 個置換及自行加入的 identity，各分解循環並做 0/1 三色背包。累加 dp[S_r][S_b][S_g]，乘群大小逆元。
proof_or_invariant: 一個染色被置換固定恰等價於每個循環單色，DP 對每循環三選一，因此精確計數固定染色。題目保證給定置換連同 identity 構成群；Burnside 引理給軌道數為固定點總和除以群大小。P>m+1 且為質數，除法可用逆元。

cpp_solution: |
  #include <algorithm>
  #include <iostream>
  #include <vector>
  using namespace std;
  int power_mod(int base, int exponent, int modulus) {
      int result = 1;
      while (exponent > 0) {
          if ((exponent & 1) != 0) result = result * base % modulus;
          base = base * base % modulus;
          exponent >>= 1;
      }
      return result;
  }
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int red, blue, green, shuffle_count, modulus;
      cin >> red >> blue >> green >> shuffle_count >> modulus;
      const int n = red + blue + green;
      vector<vector<int>> permutations(static_cast<size_t>(shuffle_count) + 1U,
                                       vector<int>(static_cast<size_t>(n)));
      for (int i = 0; i < n; ++i) permutations[0][static_cast<size_t>(i)] = i;
      for (int index = 1; index <= shuffle_count; ++index)
          for (int &destination : permutations[static_cast<size_t>(index)]) {
              cin >> destination;
              --destination;
          }
      const int blue_size = blue + 1;
      const int green_size = green + 1;
      const int total_states = (red + 1) * blue_size * green_size;
      const auto state_id = [&](int r, int b, int g) {
          return (r * blue_size + b) * green_size + g;
      };
      int fixed_sum = 0;
      for (const vector<int> &permutation : permutations) {
          vector<bool> visited(static_cast<size_t>(n), false);
          vector<int> lengths;
          for (int start = 0; start < n; ++start) {
              if (visited[static_cast<size_t>(start)]) continue;
              int length = 0;
              int current = start;
              do {
                  visited[static_cast<size_t>(current)] = true;
                  ++length;
                  current = permutation[static_cast<size_t>(current)];
              } while (current != start);
              lengths.push_back(length);
          }
          vector<int> dp(static_cast<size_t>(total_states));
          dp[0] = 1;
          for (int length : lengths) {
              vector<int> next(static_cast<size_t>(total_states));
              for (int r = 0; r <= red; ++r)
                  for (int b = 0; b <= blue; ++b)
                      for (int g = 0; g <= green; ++g) {
                          const int ways = dp[static_cast<size_t>(state_id(r, b, g))];
                          if (r + length <= red)
                              next[static_cast<size_t>(state_id(r + length, b, g))] =
                                  (next[static_cast<size_t>(state_id(r + length, b, g))] + ways) % modulus;
                          if (b + length <= blue)
                              next[static_cast<size_t>(state_id(r, b + length, g))] =
                                  (next[static_cast<size_t>(state_id(r, b + length, g))] + ways) % modulus;
                          if (g + length <= green)
                              next[static_cast<size_t>(state_id(r, b, g + length))] =
                                  (next[static_cast<size_t>(state_id(r, b, g + length))] + ways) % modulus;
                      }
              dp.swap(next);
          }
          fixed_sum = (fixed_sum + dp[static_cast<size_t>(state_id(red, blue, green))]) % modulus;
      }
      const int group_size = shuffle_count + 1;
      cout << fixed_sum * power_mod(group_size, modulus - 2, modulus) % modulus << '\n';
      return 0;
  }
