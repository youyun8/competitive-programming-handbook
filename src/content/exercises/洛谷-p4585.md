statement: |-
  n 間商店各有一種永久可買的特殊商品，之後依時間發生進貨與購物事件。
  進貨事件在商店 s 加入標價 v；購物事件給定商店區間 [l,r]、喜好密碼 x 與 d，
  可選永久商品，或最近 d 次進貨事件（含目前）中的商品，最大化 `val xor x`。
constraints:
  - n 與事件數均在官方限制內，所有輸入整數不超過 10^9
  - 商品標價為非負整數
  - d=0 時不可選任何普通進貨商品

  第一行 n、m；第二行為 n 個特殊商品標價。接下來 m 行：
  `0 s v` 表示進貨；`1 l r x d` 表示購物詢問。
output_format: 對每個購物事件輸出一行最大的 `val xor x`。

judgment: |-
  特殊商品用商店前綴 Trie 直接查詢；普通商品將最近 d 次進貨的時間窗離線拆分，
  對每個時間塊查商店區間異或最大值。
hints:
  - 先忽略進貨，永久商品可用「商店編號前綴版本」回答。
  - 一次購物對普通商品的限制是「時間連續區間 × 商店連續區間」。
  - 把時間區間拆到 O(log m) 個節點；每個節點內按商店排序後，再用前綴 Trie 限制商店範圍。
solution_outline: |-
  對特殊商品建立以商店編號為版本的可持久化 01 Trie。把每次進貨依出現順序編號；
  購物時的普通商品時間窗為 `[max(1,now-d+1),now]`，將詢問放進覆蓋此窗的線段樹節點。
  每個時間節點收集其時間段內所有進貨，按商店編號排序並建立暫時的前綴 Trie，
  以二分找到 [l,r] 對應版本差，更新詢問答案。
proof_or_invariant: |-
  線段樹的標準區間分解把時間窗拆成互斥且聯集相同的節點，因此每個合格普通商品
  恰位於其中一個被查節點。節點內前綴 Trie 版本差恰保留商店編號位於 [l,r] 的商品；
  逐位貪心優先選與 x 相反位元，得到該集合最大異或值。再與永久商品答案取最大即完備。
common_errors:
  - 把 d 解讀成原始輸入行數而不是進貨時間序
  - d=0 時仍把空時間窗插入線段樹
  - 同一商店有多次進貨時用 lower_bound 截錯前綴
complexity:

cpp_solution: |
  #include <bits/stdc++.h>
  using namespace std;

  constexpr int maximum_bit = 30;

  class PersistentTrie {
  public:
      struct Node {
          array<int, 2> child{0, 0};
          int count = 0;
      };

      PersistentTrie() { nodes_.push_back(Node{}); }

      void reserve(size_t size) { nodes_.reserve(size); }

      int insert(int old_root, int value) {
          const int new_root = clone(old_root);
          int old_node = old_root;
          int new_node = new_root;
          ++nodes_[static_cast<size_t>(new_node)].count;
          for (int bit = maximum_bit; bit >= 0; --bit) {
              const int direction = (value >> bit) & 1;
              const int old_child =
                  nodes_[static_cast<size_t>(old_node)].child[
                      static_cast<size_t>(direction)];
              const int new_child = clone(old_child);
              nodes_[static_cast<size_t>(new_node)].child[
                  static_cast<size_t>(direction)] = new_child;
              old_node = old_child;
              new_node = new_child;
              ++nodes_[static_cast<size_t>(new_node)].count;
          }
          return new_root;
      }

      int maximum_xor(int left_root, int right_root, int value) const {
          if (nodes_[static_cast<size_t>(right_root)].count ==
              nodes_[static_cast<size_t>(left_root)].count) {
              return -1;
          }
          int answer = 0;
          int left_node = left_root;
          int right_node = right_root;
          for (int bit = maximum_bit; bit >= 0; --bit) {
              const int value_bit = (value >> bit) & 1;
              const int preferred = value_bit ^ 1;
              const int preferred_left =
                  nodes_[static_cast<size_t>(left_node)].child[
                      static_cast<size_t>(preferred)];
              const int preferred_right =
                  nodes_[static_cast<size_t>(right_node)].child[
                      static_cast<size_t>(preferred)];
              if (nodes_[static_cast<size_t>(preferred_right)].count >
                  nodes_[static_cast<size_t>(preferred_left)].count) {
                  answer |= 1 << bit;
                  left_node = preferred_left;
                  right_node = preferred_right;
              } else {
                  left_node = nodes_[static_cast<size_t>(left_node)].child[
                      static_cast<size_t>(value_bit)];
                  right_node = nodes_[static_cast<size_t>(right_node)].child[
                      static_cast<size_t>(value_bit)];
              }
          }
          return answer;
      }

  private:
      vector<Node> nodes_;

      int clone(int node) {
          nodes_.push_back(nodes_[static_cast<size_t>(node)]);
          return static_cast<int>(nodes_.size()) - 1;
      }
  };

  struct Stock {
      int shop;
      int value;
  };

  struct Query {
      int left;
      int right;
      int value;
      int time_left;
      int time_right;
  };

  void add_stock(int node, int l, int r, int time, int stock_id,
                 vector<vector<int>>& stocks) {
      stocks[static_cast<size_t>(node)].push_back(stock_id);
      if (l == r) { return; }
      const int mid = (l + r) / 2;
      if (time <= mid) {
          add_stock(node * 2, l, mid, time, stock_id, stocks);
      } else {
          add_stock(node * 2 + 1, mid + 1, r, time, stock_id, stocks);
      }
  }

  void add_query(int node, int l, int r, int ql, int qr, int query_id,
                 vector<vector<int>>& queries) {
      if (qr < l || r < ql) { return; }
      if (ql <= l && r <= qr) {
          queries[static_cast<size_t>(node)].push_back(query_id);
          return;
      }
      const int mid = (l + r) / 2;
      add_query(node * 2, l, mid, ql, qr, query_id, queries);
      add_query(node * 2 + 1, mid + 1, r, ql, qr, query_id, queries);
  }

  void solve_time_tree(int node, int l, int r,
                       const vector<Stock>& all_stocks,
                       const vector<Query>& all_queries,
                       vector<vector<int>>& stocks,
                       const vector<vector<int>>& queries,
                       vector<int>& answer) {
      {
          vector<int>& ids = stocks[static_cast<size_t>(node)];
          sort(ids.begin(), ids.end(), [&](int first, int second) {
              return all_stocks[static_cast<size_t>(first)].shop <
                     all_stocks[static_cast<size_t>(second)].shop;
          });
          vector<int> shops;
          shops.reserve(ids.size());
          vector<int> root(ids.size() + 1);
          PersistentTrie trie;
          trie.reserve(ids.size() * static_cast<size_t>(maximum_bit + 2) + 1);
          for (size_t i = 0; i < ids.size(); ++i) {
              const Stock& stock =
                  all_stocks[static_cast<size_t>(ids[i])];
              shops.push_back(stock.shop);
              root[i + 1] = trie.insert(root[i], stock.value);
          }
          for (const int query_id : queries[static_cast<size_t>(node)]) {
              const Query& query =
                  all_queries[static_cast<size_t>(query_id)];
              const size_t before = static_cast<size_t>(
                  lower_bound(shops.begin(), shops.end(), query.left) -
                  shops.begin());
              const size_t through = static_cast<size_t>(
                  upper_bound(shops.begin(), shops.end(), query.right) -
                  shops.begin());
              answer[static_cast<size_t>(query_id)] = max(
                  answer[static_cast<size_t>(query_id)],
                  trie.maximum_xor(root[before], root[through], query.value));
          }
      }

      if (l == r) { return; }
      const int mid = (l + r) / 2;
      solve_time_tree(node * 2, l, mid, all_stocks, all_queries,
                      stocks, queries, answer);
      solve_time_tree(node * 2 + 1, mid + 1, r, all_stocks, all_queries,
                      stocks, queries, answer);
  }

  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);

      int shop_count, event_count;
      cin >> shop_count >> event_count;
      PersistentTrie permanent_trie;
      permanent_trie.reserve(
          static_cast<size_t>(shop_count) * (maximum_bit + 2) + 1);
      vector<int> permanent_root(static_cast<size_t>(shop_count) + 1);
      for (int shop = 1; shop <= shop_count; ++shop) {
          int value;
          cin >> value;
          permanent_root[static_cast<size_t>(shop)] =
              permanent_trie.insert(
                  permanent_root[static_cast<size_t>(shop - 1)], value);
      }

      vector<Stock> stocks(1);
      vector<Query> queries;
      vector<int> answer;
      for (int event = 0; event < event_count; ++event) {
          int type;
          cin >> type;
          if (type == 0) {
              int shop, value;
              cin >> shop >> value;
              stocks.push_back({shop, value});
          } else {
              int left, right, value, days;
              cin >> left >> right >> value >> days;
              const int current_time = static_cast<int>(stocks.size()) - 1;
              queries.push_back({
                  left,
                  right,
                  value,
                  max(1, current_time - days + 1),
                  current_time,
              });
              answer.push_back(permanent_trie.maximum_xor(
                  permanent_root[static_cast<size_t>(left - 1)],
                  permanent_root[static_cast<size_t>(right)], value));
          }
      }

      const int stock_count = static_cast<int>(stocks.size()) - 1;
      if (stock_count > 0) {
          vector<vector<int>> node_stocks(
              static_cast<size_t>(stock_count) * 4 + 4);
          vector<vector<int>> node_queries(
              static_cast<size_t>(stock_count) * 4 + 4);
          for (int time = 1; time <= stock_count; ++time) {
              add_stock(1, 1, stock_count, time, time, node_stocks);
          }
          for (size_t id = 0; id < queries.size(); ++id) {
              const Query& query = queries[id];
              if (query.time_left <= query.time_right) {
                  add_query(1, 1, stock_count, query.time_left,
                            query.time_right, static_cast<int>(id),
                            node_queries);
              }
          }
          solve_time_tree(1, 1, stock_count, stocks, queries,
                          node_stocks, node_queries, answer);
      }
      for (const int result : answer) {
          cout << result << '\n';
      }
      return 0;
  }
