hints:
  - 兩圓相連當且僅當圓心距離平方不大於半徑和平方。
  - 在無權相交圖中，分別由 1、2、3 做 BFS。
  - 枚舉匯合點 v；所需燈數為 d1[v]+d2[v]+d3[v]+1，答案用 n 減去它。
solution_outline: O(n²) 建圓相交圖，做三次 BFS；取三源距離和最小的匯合點，計算可關閉數，若不存在共同可達點則 -1。
proof_or_invariant: 對固定匯合點 v，三條最短路的邊數和最小；它們的聯集連接三終端，且即使重合也不會需要更多點，標準樹邊點數關係給出所需點數 d1+d2+d3+1。任一連通保留子圖取其連接樹的分支點可反推不小於此值，故枚舉最優。
complexity: { time: 'O(n^2)', space: 'O(n^2)' }

  #include <bits/stdc++.h>
  using namespace std;
  struct Circle{long long x,y,r;};
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int tests;if(!(cin>>tests))return 0;while(tests-->0){int n;cin>>n;vector<Circle> circles(static_cast<size_t>(n));for(Circle& circle:circles)cin>>circle.x>>circle.y>>circle.r;vector<vector<int>> graph(static_cast<size_t>(n));
          for(int i=0;i<n;++i)for(int j=i+1;j<n;++j){long long dx=circles[static_cast<size_t>(i)].x-circles[static_cast<size_t>(j)].x,dy=circles[static_cast<size_t>(i)].y-circles[static_cast<size_t>(j)].y,sum=circles[static_cast<size_t>(i)].r+circles[static_cast<size_t>(j)].r;if(dx*dx+dy*dy<=sum*sum){graph[static_cast<size_t>(i)].push_back(j);graph[static_cast<size_t>(j)].push_back(i);}}
          auto bfs=[&](int source){vector<int> dist(static_cast<size_t>(n),n+1);queue<int> queue_nodes;dist[static_cast<size_t>(source)]=0;queue_nodes.push(source);while(!queue_nodes.empty()){int u=queue_nodes.front();queue_nodes.pop();for(int v:graph[static_cast<size_t>(u)])if(dist[static_cast<size_t>(v)]>n){dist[static_cast<size_t>(v)]=dist[static_cast<size_t>(u)]+1;queue_nodes.push(v);}}return dist;};
          array<vector<int>,3> dist{bfs(0),bfs(1),bfs(2)};int minimum=3*n+1;for(int i=0;i<n;++i)minimum=min(minimum,dist[0][static_cast<size_t>(i)]+dist[1][static_cast<size_t>(i)]+dist[2][static_cast<size_t>(i)]);if(minimum>3*n)cout<<"-1\n";else cout<<n-minimum-1<<'\n';
      }
  }

三個指定點的最小連接子圖可由一個匯合點刻畫，因此三次單源最短路便足夠。
