cpp_solution: |
  #include <algorithm>
  #include <iostream>
  #include <vector>
  using namespace std;
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(nullptr);
      int n;
      while (cin >> n) {
          vector<long long> value(static_cast<size_t>(n + 1));
          for (int i = 1; i <= n; ++i) cin >> value[i];
          vector<vector<int>> children(static_cast<size_t>(n + 1));
          vector<char> has_parent(static_cast<size_t>(n + 1));
          for (int i = 1; i < n; ++i) {
              int employee, boss;
              cin >> employee >> boss;
              children[boss].push_back(employee);
              has_parent[employee] = 1;
          }
          int root = 1;
          while (has_parent[root] != 0) ++root;
          vector<long long> skip(static_cast<size_t>(n + 1));
          vector<long long> take = value;
          auto dfs = [&](auto&& self, int u) -> void {
              for (int v : children[u]) {
                  self(self, v);
                  take[u] += skip[v];
                  skip[u] += max(skip[v], take[v]);
              }
          };
          dfs(dfs, root);
          cout << max(skip[root], take[root]) << '\n';
      }
  }
