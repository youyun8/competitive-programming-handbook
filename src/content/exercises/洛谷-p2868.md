  - 二分候選平均值 mid，把每條邊權改成 fun[u]-mid×time。

solution_outline: 在 0 到最大收益間二分 60 次；每次以 SPFA 最長路判斷轉換圖是否有正環，依結果調整上下界。
proof_or_invariant: 對任一環，轉換權總和為總收益-mid×總時間，其正負恰判斷原比值是否大於 mid。故「存在正環」對 mid 單調，二分收斂到最大環比值。

  #include <bits/stdc++.h>
  using namespace std;
  struct Edge{int to,time;};
  int main(){
      ios::sync_with_stdio(false);cin.tie(nullptr);
      int n,m;if(!(cin>>n>>m))return 0;vector<double> fun(static_cast<size_t>(n));double high=0;for(double& x:fun){cin>>x;high=max(high,x);}vector<vector<Edge>> graph(static_cast<size_t>(n));for(int i=0;i<m;++i){int u,v,t;cin>>u>>v>>t;graph[static_cast<size_t>(u-1)].push_back({v-1,t});}
      auto positive_cycle=[&](double average){vector<double> dist(static_cast<size_t>(n),0);vector<int> edge_count(static_cast<size_t>(n),0);vector<char> in_queue(static_cast<size_t>(n),1);queue<int> pending;for(int i=0;i<n;++i)pending.push(i);
          while(!pending.empty()){int u=pending.front();pending.pop();in_queue[static_cast<size_t>(u)]=0;for(const Edge& e:graph[static_cast<size_t>(u)]){double candidate=dist[static_cast<size_t>(u)]+fun[static_cast<size_t>(u)]-average*e.time;if(candidate<=dist[static_cast<size_t>(e.to)]+1e-12)continue;dist[static_cast<size_t>(e.to)]=candidate;edge_count[static_cast<size_t>(e.to)]=edge_count[static_cast<size_t>(u)]+1;if(edge_count[static_cast<size_t>(e.to)]>=n)return true;if(!in_queue[static_cast<size_t>(e.to)]){in_queue[static_cast<size_t>(e.to)]=1;pending.push(e.to);}}}return false;};
      double low=0;for(int iteration=0;iteration<60;++iteration){double middle=(low+high)/2;if(positive_cycle(middle))low=middle;else high=middle;}cout<<fixed<<setprecision(2)<<low<<'\n';
  }
